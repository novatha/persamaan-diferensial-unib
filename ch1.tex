\section{Pengantar: PDB vs PDP}
\begin{frame}{Persamaan Diferensial Biasa (PDB)}
  \begin{block}{Definisi}
    Persamaan yang mengandung turunan dari satu atau lebih variabel tak bebas terhadap \textbf{satu variabel bebas tunggal} (misalnya: waktu $t$).
  \end{block}
  \vspace{0.3cm}
  Dalam Teknik Elektro, PDB sering muncul pada sistem terpusat (\textit{lumped systems}).
  \vspace{0.2cm}
  \begin{itemize}
      \item Pengosongan kapasitor: $R\frac{dq}{dt} + \frac{1}{C}q = 0$
      \item Sistem osilasi RLC: $L\frac{d^2i}{dt^2} + R\frac{di}{dt} + \frac{1}{C}i = 0$
  \end{itemize}
\end{frame}

\begin{frame}{Persamaan Diferensial Parsial (PDP)}
  \begin{exampleblock}{Definisi}
    Persamaan yang mengandung turunan parsial terhadap \textbf{dua atau lebih variabel bebas} (misalnya: koordinat ruang $x, y, z$ dan waktu $t$).
  \end{exampleblock}
  \vspace{0.3cm}
  Dalam Teknik Elektro, PDP muncul pada sistem terdistribusi (\textit{distributed systems}).
  \vspace{0.2cm}
  \begin{itemize}
      \item Persamaan Laplace (Elektrostatik 2D): $\frac{\partial^2 V}{\partial x^2} + \frac{\partial^2 V}{\partial y^2} = 0$
      \item Persamaan Gelombang (Transmisi Sinyal): $\frac{\partial^2 E}{\partial x^2} = \mu \epsilon \frac{\partial^2 E}{\partial t^2}$
  \end{itemize}
\end{frame}

\begin{frame}{Kuis Interaktif: PDB atau PDP?}
    \textbf{Tebak Persamaan Berikut:}
    \vspace{0.5cm}
    \begin{enumerate}
        \item $\frac{d^3y}{dx^3} - 4x\frac{dy}{dx} = e^x$ \pause \textcolor{unibblue}{\textbf{\quad $\rightarrow$ PDB (Hanya satu variabel bebas: x)}} \pause
        \vspace{0.4cm}
        \item $\frac{\partial u}{\partial t} = \alpha^2 \left( \frac{\partial^2 u}{\partial x^2} + \frac{\partial^2 u}{\partial y^2} \right)$ \pause \textcolor{unibgreen}{\textbf{\quad $\rightarrow$ PDP (Variabel bebas: t, x, y)}} \pause
        \vspace{0.4cm}
        \item $L \frac{di}{dt} + R i = V_0 \sin(\omega t)$ \pause \textcolor{unibblue}{\textbf{\quad $\rightarrow$ PDB (Hanya variabel bebas: t)}}
    \end{enumerate}
\end{frame}

\section{Klasifikasi: Orde \& Linieritas}
\begin{frame}{Klasifikasi: Orde}
    \begin{block}{Konsep Orde (Tingkat)}
        Orde suatu persamaan diferensial ditentukan oleh \textbf{turunan tertinggi} yang terdapat dalam persamaan tersebut.
    \end{block}
    \pause
    \vspace{0.5cm}
    \textbf{Contoh Analisis:}
    \begin{itemize}
        \item $y'' + 3y' + 2y = 0$ \pause \textcolor{red}{$\rightarrow$ Orde 2} (karena ada turunan kedua $y''$) \pause
        \item $\left(\frac{dy}{dt}\right)^{\alert{3}} + y = 0$ \pause \textcolor{red}{$\rightarrow$ Orde 1} (turunan tertingginya hanya turunan pertama, sedangkan pangkat 3 adalah "derajat", bukan orde!)
    \end{itemize}
\end{frame}

\begin{frame}{Klasifikasi: Linieritas}
    \begin{block}{Syarat Linieritas}
        Suatu PDB disebut linier jika:
        \begin{enumerate}
            \item Variabel terikat $y$ dan semua turunannya hanya berpangkat 1.
            \item Tidak ada perkalian antara $y$ dengan turunannya (misal $y \frac{dy}{dx}$).
            \item Tidak ada fungsi transendental yang memuat $y$ (misal $\sin(y)$ atau $e^y$).
        \end{enumerate}
    \end{block}
    \pause
    \vspace{0.3cm}
    \textbf{Identifikasi:}
    \begin{itemize}
        \item $\frac{d^2x}{dt^2} + 5x = \sin(t)$ \pause \textcolor{unibgreen}{\textbf{$\rightarrow$ Linier}} ($\sin(t)$ fungsi dari variabel bebas $t$, bukan $x$). \pause
        \item $\frac{dy}{dx} + y^2 = 0$ \pause \textcolor{red}{\textbf{$\rightarrow$ Non-Linier}} (karena ada suku $y^2$). \pause
        \item $\cos(x)\frac{dy}{dx} = y$ \pause \textcolor{unibgreen}{\textbf{$\rightarrow$ Linier}}.
    \end{itemize}
\end{frame}

\section{Solusi Umum, Khusus \& IVP}
\begin{frame}{Konsep Solusi Umum vs Khusus}
  \begin{columns}
  \column{0.5\textwidth}
  \begin{block}{Solusi Umum}
    Solusi yang masih memuat konstanta integrasi sembarang ($C$). Merepresentasikan himpunan/keluarga kurva. \\
    Misal PDB: $y' = y \implies y(t) = C e^t$.
  \end{block}
  \pause
  \begin{exampleblock}{Solusi Khusus}
    Jika diberikan satu kondisi awal (\textit{Initial Condition}), kita bisa mencari kurva unik yang melewati titik tersebut.
  \end{exampleblock}

  \column{0.5\textwidth}
  \begin{center}
    \begin{tikzpicture}[scale=0.8]
      % Draw axes
      \draw[->] (-0.5,0) -- (3,0) node[right] {$t$};
      \draw[->] (0,-0.5) -- (0,4) node[above] {$y(t)$};
      % Curves
      \draw[domain=0:2, smooth, variable=\t, blue] plot ({\t}, {0.5*exp(\t)});
      \draw[domain=0:1.3, smooth, variable=\t, blue] plot ({\t}, {1.0*exp(\t)});
      \draw[domain=0:0.8, smooth, variable=\t, red, thick] plot ({\t}, {2.0*exp(\t)});
      \draw[domain=0:0.5, smooth, variable=\t, blue] plot ({\t}, {3.0*exp(\t)});
      
      \fill[black] (0,2) circle (2pt) node[left] {$(0,2)$};
      \node[red] at (2, 3) {$C=2$ (Sol. Khusus)};
    \end{tikzpicture}
  \end{center}
  \end{columns}
\end{frame}

\begin{frame}{Initial Value Problem (IVP): Contoh Hitungan}
    Diberikan suatu PDB: $\frac{dy}{dt} = -3y$
    \vspace{0.2cm}
    \begin{enumerate}
        \item \textbf{Cari Solusi Umum (Diketahui dari modul sebelumnya):} \pause \\
        $y(t) = C e^{-3t}$ \pause
        \vspace{0.2cm}
        \item \textbf{Terapkan Syarat Awal (IVP):} \\
        Diketahui pada saat $t=0$, nilai $y = 10$ atau $y(0) = 10$. \pause
        \vspace{0.2cm}
        \item \textbf{Substitusi ke Solusi Umum:} \\
        $10 = C e^{-3(0)}$ \pause \\
        $10 = C e^0 \implies 10 = C (1) \implies C = 10$ \pause
        \vspace{0.2cm}
        \item \textbf{Tuliskan Solusi Khusus Akhir:} \\
        \textcolor{unibblue}{\textbf{$y(t) = 10 e^{-3t}$}}
    \end{enumerate}
\end{frame}

\section{Pemodelan Fisis: Hukum Kirchhoff}
\begin{frame}{Pemodelan Fisis: Rangkaian RL}
    Dalam Teknik Elektro, dinamika rangkaian dimodelkan dengan \textbf{Hukum Tegangan Kirchhoff (KVL)}: $\sum V = 0$.
    \begin{columns}
        \column{0.4\textwidth}
        \begin{tikzpicture}[scale=0.8, transform shape, thick]
            \draw (0,0) -- (0,2) -- (1,2);
            \draw (1,2) node[above]{R} -- (2,2) -- (2,2);
            \draw (1,2) -- (1.1,2.2) -- (1.3,1.8) -- (1.5,2.2) -- (1.7,1.8) -- (1.9,2.2) -- (2,2);
            \draw (2,2) -- (3,2);
            \draw (3,2) node[above]{L} arc (180:0:0.25) arc (180:0:0.25) arc (180:0:0.25);
            \draw (4.5,2) -- (5,2) -- (5,0) -- (0,0);
            \draw (0,1) circle (0.3) node {$V$};
            \draw (0,0) -- (0,0.7);
            \draw (0,1.3) -- (0,2);
        \end{tikzpicture}

        \column{0.6\textwidth}
        Hukum KVL untuk \textit{loop} tertutup di samping:
        \begin{equation}
            V_R + V_L = V(t)
        \end{equation}
        \pause
        Substitusi persamaan karakteristik komponen:
        \begin{itemize}
            \item $V_R = i(t) R$ (Hukum Ohm)
            \item $V_L = L \frac{di(t)}{dt}$ (Hukum Faraday/Induksi)
        \end{itemize}
        \pause
        \begin{alertblock}{Model Matematika Akhir (PDB Orde 1)}
        \begin{equation}
            L\frac{di}{dt} + R i = V(t)
        \end{equation}
        \end{alertblock}
    \end{columns}
\end{frame}
