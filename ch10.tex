% FRAME 3: Peta Kurikulum & Filosofi Pemisahan Variabel
% -------------------------------------------------------------
\begin{frame}{Peta Kurikulum: Jembatan PDP Menuju PDB Independen}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Tantangan Matematis PDP}
        \small
        \begin{itemize}
          \item Menghubungkan turunan ruang $x$ dan waktu $t$ simultan ($u_t = \alpha u_{xx}$).
          \item Tidak dapat diselesaikan dengan faktor integrasi PDB biasa.
          \item Membutuhkan pemisahan dependensi ruang dari waktu!
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Strategi Pemisahan Variabel (Fourier)}
        \small
        \begin{itemize}
          \item \textbf{1. Ansatz:} $u(x,t) = X(x) \cdot T(t)$.
          \item \textbf{2. Separasi:} $\frac{T'}{\alpha T} = \frac{X''}{X} = -\lambda$.
          \item \textbf{3. BVP Spasial:} Nilai \& fungsi eigen.
          \item \textbf{4. Superposisi:} Jumlahan deret tak hingga untuk memenuhi syarat awal $u(x,0)$.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.05cm}
  \begin{exampleblock}{Prinsip Superposisi Linier}
    \small
    Untuk operator linier homogen $\mathcal{L}(u) = 0$, kombinasi linier solusi eigen $u(x,t) = \sum_{n=1}^\infty C_n X_n(x) T_n(t)$ adalah solusi analitis eksak.
  \end{exampleblock}
\end{frame}

% FRAME 4: Postulat Produk Fungsi & Reduksi ke Dua PDB
% -------------------------------------------------------------
\begin{frame}{Postulat Produk Fungsi (\textit{Product Ansatz}) \& Reduksi PDP}
  Tinjau persamaan difusi panas atau tegangan saluran transmisi teredam:
  \begin{equation*}
    \frac{\partial u}{\partial t} = \alpha \frac{\partial^2 u}{\partial x^2}, \quad 0 < x < L, \quad t > 0
  \end{equation*}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Substitusi Produk Fungsi}
        Asumsikan $u(x, t) = X(x) \cdot T(t)$.\\
        Turunan parsial berubah menjadi turunan total:
        \begin{equation*}
          \frac{\partial u}{\partial t} = X(x) \frac{dT}{dt} = X T', \quad \frac{\partial^2 u}{\partial x^2} = X'' T
        \end{equation*}
        Substitusi ke persamaan asal:
        \begin{equation*}
          X(x) T'(t) = \alpha X''(x) T(t)
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Pemisahan Variabel Mandiri}
        Bagi kedua ruas dengan $\alpha X(x) T(t)$:
        \begin{equation*}
          \frac{T'(t)}{\alpha T(t)} = \frac{X''(x)}{X(x)} = -\lambda
        \end{equation*}
        \small
        Ruas kiri murni fungsi waktu $t$, ruas kanan murni fungsi posisi $x$.
        Kesamaan ini hanya berlaku jika keduanya bernilai \textbf{konstan} ($-\lambda$).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 5: Pemilihan Tanda Konstanta Pemisahan & Stabilitas Energi
% -------------------------------------------------------------
\begin{frame}{Pemilihan Tanda Konstanta Pemisahan: Justifikasi Fisika}
  Mengapa kita memilih konstanta separasi negatif ($-\lambda = -k^2 < 0$)?
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Kasus 1: Positif ($+k^2 > 0$)}
        \small
        $T'(t) = \alpha k^2 T(t)$\\
        $T(t) = C e^{+\alpha k^2 t}$\\
        \textbf{Fisika:} Temperatur/energi meledak tak hingga saat $t \to \infty$ ($e^{\infty} = \infty$).\\
        \textbf{Status:} \color{red}\textbf{Ditolak (Non-fisik!)}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{block}{Kasus 2: Nol ($k = 0$)}
        \small
        $T'(t) = 0 \implies T(t) = C_0$\\
        $X''(x) = 0 \implies X(x) = Ax + B$\\
        \textbf{Fisika:} Distribusi temperatur linier keadaan tunak (\textit{steady-state}) tanpa redaman.\\
        \textbf{Status:} Hanya relevan pada Neumann ($n=0$).
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Kasus 3: Negatif ($-k^2 < 0$)}
        \small
        $T'(t) = -\alpha k^2 T(t)$\\
        $T(t) = C e^{-\alpha k^2 t}$\\
        $X''(x) + k^2 X(x) = 0$\\
        $X(x) = A\cos(kx) + B\sin(kx)$\\
        \textbf{Fisika:} Energi disipatif stabil meluruh seiring waktu.\\
        \textbf{Status:} \color{unibgreen}\textbf{Diterima Secara Fisis!}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 6: Tiga Tipe Syarat Batas Kanonik
% -------------------------------------------------------------
\begin{frame}{Klasifikasi Fisis Syarat Batas (\textit{Boundary Conditions})}
  Syarat batas mendikte bagaimana sistem berinteraksi dengan lingkungan sekitarnya pada $x = 0$ dan $x = L$:
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \begin{block}{Dirichlet (Nilai Terikat)}
        \small
        \textbf{Definisi:} $u(0, t) = U_1, \; u(L, t) = U_2$.\\
        \textbf{Elektro:} Potensial pelat ditanahkan ($V = 0$) atau dicatu sumber konstan ($V = V_0$).\\
        \textbf{Termal:} Ujung konduktor dijaga pada temperatur konstan oleh \textit{heat sink masif}.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Neumann (Fluks Terisolasi)}
        \small
        \textbf{Definisi:} $\left.\frac{\partial u}{\partial x}\right|_0 = 0, \; \left.\frac{\partial u}{\partial x}\right|_L = 0$.\\
        \textbf{Elektro:} Saluran transmisi ujung terbuka ($i = 0$), atau medan listrik ortogonal batas isolator sempurna.\\
        \textbf{Termal:} Insulasi termal adiabatik sempurna (tanpa kebocoran fluks kalor).
      \end{alertblock}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Robin (Campuran / Disipatif)}
        \small
        \textbf{Definisi:} $\alpha u + \beta \frac{\partial u}{\partial x} = 0$.\\
        \textbf{Elektro:} Saluran transmisi ditutup beban impedansi terminasi ($Z_L$).\\
        \textbf{Termal:} Pendinginan konveksi udara Newton pada permukaan kabel ($\dot{q} = h(T - T_\infty)$).
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 7: Nilai Eigen & Fungsi Eigen Ortogonal
% -------------------------------------------------------------
\begin{frame}{Masalah Nilai Eigen Spasial Sturm-Liouville}
  \begin{block}{Persamaan Diferensial Spasial Homogen}
    Untuk batas Dirichlet homogen: $X''(x) + k^2 X(x) = 0$ dengan $X(0) = 0$ dan $X(L) = 0$.
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Penentuan Nilai Eigen $\lambda_n$}
        \small
        Solusi sinusoidal: $X(x) = A \cos(kx) + B \sin(kx)$.\\[2pt]
        Terapkan syarat batas:
        \begin{itemize}
          \item $X(0) = A(1) = 0 \implies \mathbf{A = 0}$
          \item $X(L) = B \sin(kL) = 0 \implies \mathbf{k_n = \frac{n\pi}{L}}$
        \end{itemize}
        Nilai eigen kuadratik: $\lambda_n = (n\pi/L)^2, \; n = 1, 2, \dots$
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Fungsi Eigen Ortogonal $X_n(x)$}
        \small
        Setiap nilai eigen $k_n$ menghasilkan fungsi eigen:
        \begin{equation*}
          \mathbf{X_n(x) = \sin\left(\frac{n\pi x}{L}\right)}
        \end{equation*}
        \textbf{Sifat Fisik Ragam Harmonis:}
        \begin{itemize}
          \item $n = 1$: Ragam fundamental (setengah gelombang).
          \item $n = 2$: Nada atas pertama (satu gelombang penuh).
          \item Saling ortogonal pada selang $[0, L]$.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contoh Terhitung: Masalah Nilai Batas Dirichlet}
  \begin{exampleblock}{Masalah Difusi 1D: $u_t = 4 u_{xx}$ pada $0 < x < \pi$, $t > 0$}
    Syarat batas: $u(0, t) = u(\pi, t) = 0$. Syarat awal: $u(x, 0) = 3\sin(x) - 7\sin(3x) + 2\sin(5x)$.
  \end{exampleblock}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penurunan Fungsi Eigen}
        \small
        \textbf{1. Nilai Eigen Spasial ($L = \pi$):}
        $k_n = n, \quad X_n(x) = \sin(nx)$.\\[2pt]
        \textbf{2. Solusi Temporal ($\alpha = 4$):}
        $T_n'(t) = -4n^2 T_n(t) \implies T_n(t) = e^{-4n^2 t}$.\\[2pt]
        \textbf{3. Superposisi Solusi Umum:}
        \begin{equation*}
          u(x, t) = \sum_{n=1}^{\infty} C_n \sin(nx) e^{-4n^2 t}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Pencocokan Koefisien \& Solusi}
        \small
        \textbf{4. Syarat Awal ($t = 0$):}
        \begin{align*}
          u(x, 0) &= \sum_{n=1}^{\infty} C_n \sin(nx) \\
                  &= 3\sin(x) - 7\sin(3x) + 2\sin(5x)
        \end{align*}
        Inspeksi: $C_1 = 3, \; C_3 = -7, \; C_5 = 2, \; C_{\text{lain}} = 0$.\\[2pt]
        \textbf{5. Solusi Analitis Eksak:}
        \begin{align*}
          u(x, t) &= 3\sin(x)e^{-4t} - 7\sin(3x)e^{-36t} \\
                  &\quad + 2\sin(5x)e^{-100t}
        \end{align*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 10: Kasus Batas Neumann (Ujung Terisolasi)
% -------------------------------------------------------------
\begin{frame}{Kasus Batas Neumann: Ujung Batang Terisolasi}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Syarat Batas Fluks Nol ($u_x = 0$)}
        \small
        Batas terisolasi: $X'(0) = 0$ dan $X'(L) = 0$.\\
        Turunan fungsi umum $X(x) = A\cos(kx) + B\sin(kx)$:
        \begin{equation*}
          X'(x) = -k A \sin(kx) + k B \cos(kx)
        \end{equation*}
        \begin{itemize}
          \item $X'(0) = k B (1) = 0 \implies \mathbf{B = 0}$
          \item $X'(L) = -k A \sin(kL) = 0 \implies \mathbf{k_n = \frac{n\pi}{L}}$
        \end{itemize}
        Untuk $n = 0$: $k_0 = 0 \implies \mathbf{X_0(x) = 1}$ (\textbf{Ragam Nol}).
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Deret Superposisi Kosinus Neumann}
        \small
        Solusi lengkap memuat konstanta nilai rata-rata:
        \begin{equation*}
          u(x, t) = A_0 + \sum_{n=1}^{\infty} A_n \cos\left(\frac{n\pi x}{L}\right) e^{-\alpha (n\pi/L)^2 t}
        \end{equation*}
        \textbf{Makna Fisis Kestabilan Termal:}
        \begin{itemize}
          \item Saat $t \to \infty$, seluruh suku $e^{-\dots t} \to 0$.
          \item Solusi tunak: $u(x, \infty) = A_0$ (temperatur merata sempurna di seluruh batang akibat isolasi sempurna tanpa energi keluar).
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 11: Aplikasi 2D Elektrostatika: Persamaan Laplace
% -------------------------------------------------------------
\begin{frame}{Ekspansi Pemisahan Variabel 2D: Persamaan Laplace}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{block}{BVP Elektrostatik dalam Rongga Pelat}
        \small
        Cari potensial $\Phi(x,y)$ di dalam daerah persegi $0 < x < a, \, 0 < y < b$:
        \begin{equation*}
          \frac{\partial^2 \Phi}{\partial x^2} + \frac{\partial^2 \Phi}{\partial y^2} = 0
        \end{equation*}
        Syarat batas: 3 dinding ditanahkan ($\Phi(0,y) = \Phi(a,y) = \Phi(x,0) = 0$) dan atap diberi tegangan $\Phi(x,b) = V_0$.
      \end{block}
      \begin{alertblock}{Pemisahan $\Phi(x,y) = X(x) Y(y)$}
        \small
        $\frac{X''}{X} = -\frac{Y''}{Y} = -k_n^2 \implies X_n(x) = \sin\left(\frac{n\pi x}{a}\right)$\\
        $Y_n(y) = \sinh\left(\frac{n\pi y}{a}\right)$ (karena $Y(0) = 0$).
      \end{alertblock}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{Solusi Superposisi Analitis}
        \small
        Penerapan syarat atap dengan deret Fourier:
        \begin{equation*}
          \Phi(x, y) = \frac{4V_0}{\pi} \sum_{k=1}^{\infty} \frac{\sin\left(\frac{(2k-1)\pi x}{a}\right) \sinh\left(\frac{(2k-1)\pi y}{a}\right)}{(2k-1) \sinh\left(\frac{(2k-1)\pi b}{a}\right)}
        \end{equation*}
        \textbf{Wawasan Rekayasa:}
        Bentuk fungsi hiperbolik $\sinh(y)$ mencerminkan konsentrasi medan listrik di dekat pelat tegangan tinggi.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

      \footnotesize
      \textbf{Kasus Rekayasa:} Pada pemisahan variabel $u(x,t) = X(x)T(t)$ untuk persamaan termal, mengapa konstanta pemisah $k$ wajib dipilih bernilai negatif ($-\lambda^2$)?
      \vspace{0.05cm}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{[A]} Konstanta positif dilarang dalam kalkulus diferensial.
        \item \textbf{[B]} Konstanta positif menghasilkan solusi $T(t) = C e^{+\lambda^2 t} \to \infty$, yang melanggar hukum termodinamika bahwa panas harus meluruh stabil!
        \item \textbf{[C]} Konstanta negatif mempercepat perhitungan determinan.
        \item \textbf{[D]} Nilai positif hanya berlaku untuk gelombang suara.
      \end{itemize}
    \end{alertblock}

        \item \textbf{C5 (Evaluasi):} Evaluasi konvergensi deret Fourier solusi transien terhadap kondisi awal tak rata.
        \item \textbf{C6 (Desain):} Rancang panjang busbar gardu induk agar kenaikan suhu akibat arus nominal $3000\text{ A}$ tidak melampaui $90^\circ\text{C}$!
