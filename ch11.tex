\begin{frame}{Peta Kurikulum: Batas Rangkaian Terpusat vs Terdistribusi}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Kapan KVL/KCL Biasa Tidak Berlaku?}
        \small
        \begin{itemize}
          \item Pada frekuensi rendah ($50\text{ Hz}$), panjang gelombang $\lambda = c/f = 6.000\text{ km}$. Sirkuit laboratorium ($\ell \sim 1\text{ m}$) mematuhi parameter terpusat.
          \item Namun pada surja petir (frekuensi ekivalen $\sim 1\text{ MHz}$, $\lambda = 300\text{ m}$) atau pulsa digital pita lebar, panjang saluran transmisi $\ell \gg \lambda / 10$.
          \item Fasa tegangan bervariasi di sepanjang kawat!
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Konsekuensi Fisik Saluran Terdistribusi}
        \small
        \begin{itemize}
          \item Sinyal merambat dengan kecepatan berhingga $v_p \le c$.
          \item Muncul fenomena waktu tunda transit: $t_d = \ell / v_p$.
          \item Terjadi pantulan gelombang jika ujung saluran tidak tersuai (\textit{unmatched termination}).
          \item Diformulasikan secara eksak oleh \textbf{Persamaan Telegrafer}!
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{0.05cm}
  \begin{exampleblock}{Prinsip Gelombang Elektromagnetik Transversal (TEM)}
    \small
    Energi tidak merambat di dalam inti tembaga, melainkan mengalir di ruang dielektrik antara kedua konduktor melalui medan listrik $\mathbf{E}$ dan medan magnet $\mathbf{H}$.
  \end{exampleblock}
\end{frame}

% FRAME 4: Model Elemen Mikro dx & Persamaan Telegrafer
% -------------------------------------------------------------
\begin{frame}{Penurunan Model Elemen Mikro $\Delta x$ Saluran Transmisi}
  Tinjau segmen saluran mikro $\Delta x$ dengan parameter terdistribusi $L'$ (H/m) dan $C'$ (F/m):
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Hukum Tegangan Kirchhoff (KVL)}
        \small
        Jatuh tegangan pada induktansi seri:
        \begin{equation*}
          v(x+\Delta x, t) - v(x, t) = -L' \Delta x \frac{\partial i(x, t)}{\partial t}
        \end{equation*}
        Bagi dengan $\Delta x \to 0$ (Hukum Faraday):
        \begin{equation*}
          \mathbf{\frac{\partial v}{\partial x} = -L' \frac{\partial i}{\partial t}}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Hukum Arus Kirchhoff (KCL)}
        \small
        Arus bocor melalui kapasitansi paralel:
        \begin{equation*}
          i(x+\Delta x, t) - i(x, t) = -C' \Delta x \frac{\partial v(x+\Delta x, t)}{\partial t}
        \end{equation*}
        Bagi dengan $\Delta x \to 0$ (Arus Pergeseran Maxwell):
        \begin{equation*}
          \mathbf{\frac{\partial i}{\partial x} = -C' \frac{\partial v}{\partial t}}
        \end{equation*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 5: Reduksi ke Persamaan Gelombang 1D
% -------------------------------------------------------------
\begin{frame}{Eliminasi Silang Menuju Persamaan Gelombang 1D}
  Dua persamaan telegrafer orde 1 berkopel dapat diubah menjadi satu PDP orde 2 murni:
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Eliminasi Arus $i(x, t)$}
        \small
        Diferensialkan pers. tegangan thd $x$:
        \begin{equation*}
          \frac{\partial^2 v}{\partial x^2} = -L' \frac{\partial^2 i}{\partial x \partial t}
        \end{equation*}
        Substitusi $\frac{\partial i}{\partial x} = -C' \frac{\partial v}{\partial t}$:
        \begin{equation*}
          \frac{\partial^2 v}{\partial x^2} = -L' \frac{\partial}{\partial t}\left(-C'\frac{\partial v}{\partial t}\right) = L'C' \frac{\partial^2 v}{\partial t^2}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Bentuk Kanonik Persamaan Gelombang}
        \small
        \begin{equation*}
          \mathbf{\frac{\partial^2 v}{\partial x^2} = \frac{1}{v_p^2} \frac{\partial^2 v}{\partial t^2}}
        \end{equation*}
        dengan \textbf{Kecepatan Rambat Fasa}:
        \begin{equation*}
          \mathbf{v_p = \frac{1}{\sqrt{L'C'}} = \frac{c}{\sqrt{\mu_r \epsilon_r}}}
        \end{equation*}
        Pada kawat udara: $v_p \approx c = 3 \times 10^8\text{ m/s}$. Pada kabel dielektrik padat: $v_p \approx 2 \times 10^8\text{ m/s}$.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 6: Solusi d'Alembert Gelombang Berjalan
% -------------------------------------------------------------
\begin{frame}{Solusi Perjalanan Gelombang d'Alembert}
  \begin{block}{Teorema d'Alembert (1747)}
    \small
    Solusi umum dari $\frac{\partial^2 v}{\partial x^2} - \frac{1}{v_p^2} \frac{\partial^2 v}{\partial t^2} = 0$ adalah superposisi dari dua gelombang sembarang yang mempertahankan bentuknya:
    \begin{equation*}
      \mathbf{v(x, t) = f(x - v_p t) + g(x + v_p t) = v^+(x - v_p t) + v^-(x + v_p t)}
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Gelombang Maju: $v^+(x - v_p t)$}
        \small
        \begin{itemize}
          \item Merambat ke arah $+x$ (dari sumber menuju beban).
          \item Nilai gelombang konstan sepanjang garis karakteristik: $x - v_p t = \text{konstan}$.
          \item Membawa daya aktif dari generator ke beban.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Gelombang Pantul: $v^-(x + v_p t)$}
        \small
        \begin{itemize}
          \item Merambat ke arah $-x$ (dari beban kembali ke sumber).
          \item Garis karakteristik: $x + v_p t = \text{konstan}$.
          \item Timbul akibat diskontinuitas impedansi pada ujung saluran transmisi.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 7: Impedansi Karakteristik Saluran Transmisi
% -------------------------------------------------------------
\begin{frame}{Impedansi Karakteristik $Z_0$ \& Rasio Gelombang Arus}
  \begin{block}{Definisi Fisis Impedansi Karakteristik}
    \small
    Rasio antara gelombang tegangan berjalan maju terhadap gelombang arus berjalan maju pada sembarang titik saluran:
    \begin{equation*}
      \mathbf{Z_0 = \frac{v^+(x, t)}{i^+(x, t)} = \sqrt{\frac{L'}{C'}}} \quad [\Omega]
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Pembalikan Tanda pada Arus Pantul}
        \small
        Karena daya gelombang mundur mengalir ke arah $-x$:
        \begin{equation*}
          \frac{v^-(x, t)}{i^-(x, t)} = -Z_0 \implies \mathbf{i^-(x, t) = -\frac{v^-(x, t)}{Z_0}}
        \end{equation*}
        Tegangan total adalah jumlahan fasa ($v^+ + v^-$), sedangkan arus total adalah pengurangan ($i^+ - i^-$).
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Nilai Tipikal Standar Industri}
        \small
        \begin{itemize}
          \item \textbf{Saluran Udara (SUTT 150 kV):} $Z_0 \approx 350 - 450\,\Omega$.
          \item \textbf{Kabel Tanah XLPE:} $Z_0 \approx 30 - 50\,\Omega$.
          \item \textbf{Kabel Koaksial RG-58:} $Z_0 = 50\,\Omega$ (RF \& Lab).
          \item \textbf{Kabel Antena TV RG-6:} $Z_0 = 75\,\Omega$.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contoh Terhitung: Solusi d'Alembert Impuls Awal}
  \begin{exampleblock}{Masalah: Pulsa awal $v(x, 0) = \frac{100}{1 + x^2}\text{ [V]}$ dan $\left.\frac{\partial v}{\partial t}\right|_{t=0} = 0$ pada $v_p = 2 \times 10^8\text{ m/s}$}
    Tentukan ekspresi gelombang berjalan eksak $v(x, t)$ dan sketsa perilaku fisik perambatannya!
  \end{exampleblock}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penurunan Rumus Integral d'Alembert}
        \begin{align*}
          v(x, t) &= \frac{\phi(x - v_p t) + \phi(x + v_p t)}{2} \\
                  &\quad + \frac{1}{2v_p}\int_{x-v_p t}^{x+v_p t} \psi(\xi)\,d\xi
        \end{align*}
        Karena kecepatan awal $\psi(\xi) = 0$, suku integral nol!
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Solusi Eksak \& Perilaku Gelombang}
        \small
        Substitusi fungsi profil awal $\phi(x)$:
        \begin{align*}
          v(x, t) &= \frac{50}{1 + (x - 2\times 10^8 t)^2} \\
                  &\quad + \frac{50}{1 + (x + 2\times 10^8 t)^2} \quad \text{[V]}
        \end{align*}
        \textbf{Interpretasi Fisik:}
        Pulsa mula-mula $100\text{ V}$ terbelah menjadi dua pulsa simetris masing-masing $50\text{ V}$ yang merambat saling menjauhi titik asal!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 10: Refleksi Gelombang pada Beban
% -------------------------------------------------------------
\begin{frame}{Refleksi Gelombang pada Terminasi Beban $Z_L$}
  Tegangan dan arus pada ujung beban ($x = L$) harus memenuhi Hukum Ohm: $v(L, t) = Z_L \cdot i(L, t)$.
  \begin{alertblock}{Koefisien Refleksi Beban (\textit{Load Reflection Coefficient})}
    \begin{equation*}
      \mathbf{\Gamma_L = \frac{V^-}{V^+} = \frac{Z_L - Z_0}{Z_L + Z_0}} \quad (-1 \le \Gamma_L \le +1)
    \end{equation*}
  \end{alertblock}
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \begin{block}{Beban Tersuai ($Z_L = Z_0$)}
        \small
        $\Gamma_L = 0, \quad V^- = 0$\\
        Seluruh energi gelombang diserap sempurna oleh beban tanpa pantulan.\\
        \textit{Ideal untuk telekomunikasi!}
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Ujung Terbuka ($Z_L = \infty$)}
        \small
        $\Gamma_L = +1, \quad V^- = +V^+$\\
        Tegangan berlipat ganda: $V_{\text{beban}} = 2 V^+$, arus terhenti nol ($I_{\text{beban}} = 0$).\\
        \textit{Bahaya surja petir!}
      \end{exampleblock}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Hubung Singkat ($Z_L = 0$)}
        \small
        $\Gamma_L = -1, \quad V^- = -V^+$\\
        Tegangan lenyap nol ($V = 0$), arus berlipat ganda: $I_{\text{beban}} = 2 I^+$.\\
        \textit{Pembalikan polaritas!}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 11: Gelombang Berdiri & VSWR
% -------------------------------------------------------------
\begin{frame}{Gelombang Berdiri (\textit{Standing Waves}) \& VSWR}
  Superposisi gelombang datang dan pantul kontinu menghasilkan pola interferensi stasioner:
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Tegangan Ekstrem pada Saluran}
        \small
        \begin{itemize}
          \item \textbf{Tegangan Maksimum ($V_{\text{max}}$):} Interferensi konstruktif:
          \begin{equation*}
            V_{\text{max}} = |V^+| (1 + |\Gamma_L|)
          \end{equation*}
          \item \textbf{Tegangan Minimum ($V_{\text{min}}$):} Interferensi destruktif:
          \begin{equation*}
            V_{\text{min}} = |V^+| (1 - |\Gamma_L|)
          \end{equation*}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Voltage Standing Wave Ratio (VSWR)}
        \small
        Parameter kualitas transmisi daya:
        \begin{equation*}
          \mathbf{\text{VSWR} = \frac{V_{\text{max}}}{V_{\text{min}}} = \frac{1 + |\Gamma_L|}{1 - |\Gamma_L|}} \quad (1 \le \text{VSWR} < \infty)
        \end{equation*}
        \begin{itemize}
          \item $\text{VSWR} = 1{,}0$: Penyesuaian sempurna (tanpa daya pantul).
          \item $\text{VSWR} > 2{,}0$: Bahaya kerusakan pemancar/insulasi.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

      \footnotesize
      \textbf{Kasus Rekayasa:} Saluran transmisi $150\text{ kV}$ memiliki ujung terbuka (*open circuit*, $Z_L \to \infty$). Ketika surja petir $V^+$ menabrak ujung saluran tersebut, tegangan sesaat pada isolator ujung saluran menjadi:
      \vspace{0.05cm}
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item \textbf{[A]} Nol karena arus tidak dapat mengalir ke luar kawat.
        \item \textbf{[B]} **Dua kali lipat** ($2V^+$) akibat pemantulan sefasa penuh ($\Gamma_L = +1$)!
        \item \textbf{[C]} Setengahnya ($V^+/2$) karena pembagian tegangan.
        \item \textbf{[D]} Berkurang drastis terserap ke udara bebas.
      \end{itemize}
    \end{alertblock}

    \column{0.46\textwidth}
    \begin{block}{Tantangan Analisis \& Desain (C4--C6)}

      \begin{itemize}\setlength{\itemsep}{2pt}

        \item \textbf{C5 (Evaluasi):} Evaluasi nilai Voltage Standing Wave Ratio (VSWR) pada ketidakcocokan beban $Z_L \ne Z_0$.
        \item \textbf{C6 (Desain):} Rancang koordinasi isolasi lightning arrester (LA) IEC 60071-1 agar tegangan pantul tidak merusak trafo!
