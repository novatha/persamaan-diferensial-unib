% FRAME 4: Hukum Fourier & Persamaan Difusi Panas 1D
% -------------------------------------------------------------
\begin{frame}{Hukum Fourier Konduksi \& Penurunan Persamaan Panas}
  Tinjau elemen konduktor mikro $\Delta x$ berpenampang $A$ dengan konduktivitas termal $k$ [W/(m$\cdot$K)]:
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Hukum Fourier Perpindahan Panas}
        \small
        Fluks kalor berbanding lurus dengan gradien spasial:
        \begin{equation*}
          q_x = -k A \frac{\partial u}{\partial x} \quad \text{[Watt]}
        \end{equation*}
        Tanda minus menandakan kalor mengalir menuruni gradien temperatur (dari panas ke dingin).
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Konservasi Energi pada Elemen $\Delta x$}
        \small
        Laju akumulasi = Kalor masuk $-$ Kalor keluar $+$ Sumber:
        \begin{equation*}
          \rho c_p A \Delta x \frac{\partial u}{\partial t} = -\frac{\partial q_x}{\partial x} \Delta x + \dot{q}_{\text{gen}} A \Delta x
        \end{equation*}
        Maka diperoleh \textbf{Persamaan Difusi 1D}:
        \begin{equation*}
          \mathbf{\frac{\partial u}{\partial t} = \alpha \frac{\partial^2 u}{\partial x^2} + \frac{\dot{q}_{\text{gen}}}{\rho c_p}}
        \end{equation*}
        dengan \textbf{Difusivitas Termal}: $\alpha = \frac{k}{\rho c_p}$ [m$^2$/s].
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 5: Dekomposisi Solusi: Tunak & Transien
% -------------------------------------------------------------
\begin{frame}{Metode Dekomposisi: Keadaan Tunak \& Transien}
  Untuk menyelesaikan PDP dengan batas non-homogen atau sumber internal $\dot{q}_{\text{gen}}$:
  \begin{block}{Postulat Superposisi Dekomposisi}
    \small
    Pecah solusi lengkap $u(x, t)$ menjadi dua bagian independen:
    \begin{equation*}
      \mathbf{u(x, t) = u_{\text{ss}}(x) + u_{\text{tr}}(x, t)}
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{1. Solusi Keadaan Tunak: $u_{\text{ss}}(x)$}
        \small
        Kondisi asimtotik saat $t \to \infty$ ($\partial u/\partial t = 0$):
        \begin{equation*}
          \alpha \frac{d^2 u_{\text{ss}}}{dx^2} + \frac{\dot{q}_{\text{gen}}}{\rho c_p} = 0 \implies \frac{d^2 u_{\text{ss}}}{dx^2} = -\frac{\dot{q}_{\text{gen}}}{k}
        \end{equation*}
        Merupakan \textbf{PDB Biasa Orde Dua}! Selesaikan dengan integrasi langsung dua kali.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{2. Solusi Transien: $u_{\text{tr}}(x, t)$}
        \small
        Memenuhi PDP homogen tanpa sumber:
        \begin{equation*}
          \frac{\partial u_{\text{tr}}}{\partial t} = \alpha \frac{\partial^2 u_{\text{tr}}}{\partial x^2}
        \end{equation*}
        Syarat batas dibuat homogen ($u_{\text{tr}} = 0$), diselesaikan dengan pemisahan variabel $\implies$ meluruh ke nol saat $t \to \infty$.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 6: Batas Dirichlet Homogen & Waktu Relaksasi
% -------------------------------------------------------------
\begin{frame}{Pendinginan Dirichlet \& Skala Waktu Relaksasi Termal}
  Jika kawat panas didinginkan dengan kedua ujungnya terikat pada suhu ambien $0^\circ\text{C}$:
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Solusi Deret Fourier Sinus Dirichlet}
        \small
        \begin{equation*}
          u(x, t) = \sum_{n=1}^{\infty} B_n \sin\left(\frac{n\pi x}{L}\right) \exp\left(-\frac{t}{\tau_n}\right)
        \end{equation*}
        dengan \textbf{Konstanta Waktu Ragam ke-$n$}:
        \begin{equation*}
          \mathbf{\tau_n = \frac{L^2}{\alpha \pi^2 n^2}}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Dominasi Ragam Fundamental ($n = 1$)}
        \small
        \begin{itemize}
          \item $\tau_1 = \frac{L^2}{\alpha \pi^2}$ (waktu pendinginan utama).
          \item $\tau_2 = \frac{\tau_1}{4}, \quad \tau_3 = \frac{\tau_1}{9}, \quad \tau_5 = \frac{\tau_1}{25}$.
          \item Seluruh riak dan ketajaman bentuk awal lenyap sangat cepat karena harmonisa tinggi meluruh sebanding kuadrat orde ($n^2$).
          \item Profil temperatur selalu berakhir menyerupai kurva sinusoidal mulus.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 7: Batas Campuran Neumann-Dirichlet
% -------------------------------------------------------------
\begin{frame}{Kondisi Batas Campuran: Insulasi \& Pendinginan Kuartal Gelombang}
  Tinjau konduktor di mana ujung $x = 0$ diisolasi sempurna ($u_x = 0$) dan ujung $x = L$ didinginkan ($u = 0$):
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penentuan Fungsi Eigen Kuartal Gelombang}
        \small
        PDB spasial: $X''(x) + k^2 X(x) = 0$.\\
        $X(x) = A\cos(kx) + B\sin(kx)$.
        \begin{itemize}
          \item Batas Neumann di $x=0$:
          $X'(0) = k B = 0 \implies \mathbf{B = 0}$.
          \item Batas Dirichlet di $x=L$:
          $X(L) = A \cos(kL) = 0 \implies k_n L = \frac{(2n-1)\pi}{2}$.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Nilai Eigen \& Fungsi Eigen}
        \small
        \begin{equation*}
          \mathbf{k_n = \frac{(2n-1)\pi}{2L}}, \quad \mathbf{\lambda_n = \left(\frac{(2n-1)\pi}{2L}\right)^2}
        \end{equation*}
        Fungsi eigen berupa kosinus kuartal gelombang:
        \begin{equation*}
          \mathbf{X_n(x) = \cos\left(\frac{(2n-1)\pi x}{2L}\right)}
        \end{equation*}
        Pola ini sering dijumpai pada inti kabel bawah tanah tertutup dengan pendinginan luar.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 9: Contoh Soal Terhitung Langkah demi Langkah
% -------------------------------------------------------------
\begin{frame}{Contoh Terhitung: Pendinginan Batang Tembaga $100^\circ\text{C} \to 50^\circ\text{C}$}
  \begin{exampleblock}{Masalah: Batang $L = 1\text{ m}$, suhu awal $100^\circ\text{C}$, batas $u(0)=u(1)=0^\circ\text{C}$, $\alpha = 10^{-4}\text{ m}^2/\text{s}$}
    Berapa waktu $t$ agar temperatur pusat batang ($x = 0{,}5\text{ m}$) turun menjadi $50^\circ\text{C}$?
  \end{exampleblock}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penurunan Deret Fourier Dirichlet}
        \small
        Koefisien Fourier ($f(x) = 100$):
        \begin{equation*}
          B_n = \frac{2}{1}\int_{0}^{1} 100\sin(n\pi x)\,dx = \begin{cases} \frac{400}{n\pi}, & n \text{ ganjil} \\ 0, & n \text{ genap} \end{cases}
        \end{equation*}
        Ekspansi deret temperatur:
        \begin{equation*}
          u(x, t) = \frac{400}{\pi}\sum_{k=1}^{\infty}\frac{\sin((2k-1)\pi x)}{2k-1} e^{-\alpha (2k-1)^2 \pi^2 t}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Evaluasi Waktu pada Titik Tengah}
        \small
        Di $x = 0{,}5$, $\sin(\pi / 2) = 1$. Ambil ragam $n = 1$:
        \begin{equation*}
          u(0{,}5, t) \approx \frac{400}{\pi} e^{-\alpha \pi^2 t} = 50^\circ\text{C}
        \end{equation*}
        Selesaikan untuk waktu $t$:
        \begin{align*}
          e^{-\alpha \pi^2 t} &= \frac{50\pi}{400} \approx 0{,}3927 \\
          t &= \frac{-\ln(0{,}3927)}{\alpha \pi^2} = \frac{0{,}9347}{(10^{-4})\pi^2} \\
          &\approx \mathbf{947\text{ s}} \approx \mathbf{15{,}8\text{ menit}} \quad \qed
        \end{align*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contoh Terhitung: Pemanasan Internal Joule Kawat Listrik}
  \begin{exampleblock}{Masalah: Kawat $L = 2\text{ m}, k = 40\text{ W/(m}\cdot\text{K)}$, $\dot{q}_{\text{gen}} = 8.000\text{ W/m}^3$, batas $u(0)=u(2)=20^\circ\text{C}$}
    Tentukan profil temperatur keadaan tunak $u_{\text{ss}}(x)$ dan temperatur puncak kawat!
  \end{exampleblock}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penyusunan PDB Keadaan Tunak}
        \small
        Kondisi mantap ($\partial u/\partial t = 0$):
        \begin{equation*}
          \frac{d^2 u_{\text{ss}}}{dx^2} = -\frac{\dot{q}_{\text{gen}}}{k} = -\frac{8.000}{40} = -200
        \end{equation*}
        Integrasi pertama: $\frac{d u_{\text{ss}}}{dx} = -200 x + C_1$.\\[2pt]
        Integrasi kedua: $u_{\text{ss}}(x) = -100 x^2 + C_1 x + C_2$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Penerapan Batas \& Puncak}
        \small
        \textbf{Syarat Batas:}
        $u_{\text{ss}}(0) = 20 \implies C_2 = 20$,\\
        $u_{\text{ss}}(2) = 20 \implies C_1 = 200$.\\[2pt]
        Profil temperatur parabola simetris:
        \begin{equation*}
          \mathbf{u_{\text{ss}}(x) = -100 x^2 + 200 x + 20} \quad [^\circ\text{C}]
        \end{equation*}
        Puncak di tengah ($x = 1{,}0\text{ m}$):
        \begin{equation*}
          u_{\text{max}} = u_{\text{ss}}(1) = \mathbf{120^\circ\text{C}} \quad \qed
        \end{equation*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
\begin{frame}{Kuat Hantar Arus (\textit{Ampacity}) \& Degradasi Isolasi}
  \begin{block}{Definisi Matematis Kemampuan Hantar Arus Konduktor}
    \small
    \textbf{Ampacity} ($I_{\text{max}}$) adalah arus kontinyu maksimum aman:
    \begin{equation*}
      \mathbf{I_{\text{max}} = \sqrt{\frac{\theta_{\text{max}} - \theta_{\text{amb}}}{R_{\text{ac}} \cdot R_{\text{th,total}}}}} \quad \text{[Ampere]}
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Bahaya Melampaui Batas Operasi}
        \small
        \begin{itemize}
          \item \textbf{Penuaan Arrhenius:} Kenaikan suhu $+10^\circ\text{C}$ memotong umur kabel $50\%$.
          \item \textbf{Water Treed:} Pelunakan XLPE mempercepat tembus lucutan parsial.
          \item \textbf{Deformasi Mekanis:} Pelemahan selubung timbal pelindung kabel.
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Tahanan Termal Tanah ($T_4$)}
        \small
        \begin{itemize}
          \item Tahanan tanah kering tinggi ($\rho_{\text{tanah}} \sim 2{,}5\text{ K}\cdot\text{m/W}$).
          \item Migrasi kelembaban (\textit{thermal runaway}) terjadi saat tanah di sekitar kabel mengering total akibat disipasi berlebih.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{Praktikum Komputasi Julia: Difusi Termal Kabel XLPE}

    \column{0.50\textwidth}
    \begin{block}{Skrip Julia: Penetrasi Panas Transien}
\begin{lstlisting}[style=juliastyle]
using Plots

r_norm = range(0, 1.0, length=200)
T_prof(r, tau) = (1.0 .- r) .* (1.0 .- exp.(-r .* 4 .- tau .* 2))
