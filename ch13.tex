\section{Penurunan Persamaan Laplace \& Poisson}
\begin{frame}{Dari Hukum Gauss ke Persamaan Laplace}
  \begin{block}{Hukum Medan Elektrostatik}
    $$ \nabla \cdot \mathbf{E} = \frac{\rho_v}{\varepsilon}, \quad \mathbf{E} = -\nabla V $$
  \end{block}
  \pause
  Substitusi medan listrik $\mathbf{E}$ ke Hukum Gauss:
  $$ \nabla \cdot (-\nabla V) = \frac{\rho_v}{\varepsilon} \implies {\color{unibblue} \nabla^2 V = -\frac{\rho_v}{\varepsilon}} \quad \text{(Persamaan Poisson)} $$
  \pause
  \begin{alertblock}{Persamaan Laplace 2D (Ruang Bebas Muatan $\rho_v = 0$)}
    $$ {\color{unibblue} \frac{\partial^2 V}{\partial x^2} + \frac{\partial^2 V}{\partial y^2} = 0} $$
  \end{alertblock}
\end{frame}

\section{Sifat Fungsi Harmonik}
\begin{frame}{Keunikan Solusi Laplace (Fungsi Harmonik)}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{Sifat Nilai Rata-rata}
        Potensial pada titik manapun $(x,y)$ selalu sama dengan nilai rata-rata potensial pada lingkaran/kotak di sekelilingnya:
        $$ V(x,y) = \frac{1}{2\pi}\int_0^{2\pi} V(R, \theta) d\theta $$
      \end{block}
    \end{column}
    \begin{column}{0.5\textwidth}
      \pause
      \begin{alertblock}{Prinsip Maksimum-Minimum}
        Fungsi harmonik tidak pernah memiliki nilai maksimum lokal atau minimum lokal di dalam domain terbuka! Nilai ekstrem hanya terjadi pada batas konduktor (*boundaries*).
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\section{Solusi Analitik Deret 2D}
\begin{frame}{Distribusi Potensial dalam Konduktor Persegi Panjang}
  Domain $0 \le x \le a$ dan $0 \le y \le b$ dengan batas:
  $V(0,y) = 0, \quad V(a,y) = 0, \quad V(x,0) = 0, \quad V(x,b) = V_0$.
  \pause
  Menggunakan pemisahan variabel $V(x,y) = X(x)Y(y)$:
  $$ X_n(x) = \sin\left(\frac{n\pi x}{a}\right), \quad Y_n(y) = \sinh\left(\frac{n\pi y}{a}\right) $$
  \pause
  \begin{alertblock}{Solusi Deret Fourier 2D}
    $$ {\color{unibblue} V(x,y) = \sum_{n=1,3,5,...}^{\infty} \frac{4V_0}{n\pi \sinh\left(\frac{n\pi b}{a}\right)} \sin\left(\frac{n\pi x}{a}\right) \sinh\left(\frac{n\pi y}{a}\right)} $$
  \end{alertblock}
\end{frame}

\section{Simulasi Julia \& Kuis}
\begin{frame}{Simulasi Numerik Relaksasi Gauss-Seidel (Julia)}
  \begin{block}{Algoritma Beda Hingga 4-Titik}
    $$ V_{i,j}^{(k+1)} = \frac{1}{4} \left( V_{i+1,j}^{(k)} + V_{i-1,j}^{(k)} + V_{i,j+1}^{(k)} + V_{i,j-1}^{(k)} \right) $$
    \begin{itemize}
      \item Menghasilkan garis ekipotensial kontur 2D.
      \item Memvisualisasikan permukaan potensial 3D (*Potential Wells*).
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Kuis Interaktif 13}
  \textbf{Soal:} Jika seluruh batas konduktor luar sebuah rongga kotak tertutup dihubungkan ke tanah ($V = 0\text{ V}$), berapa nilai potensial elektrostatik $V$ persis di tengah-tengah rongga tersebut?
  \pause
  \begin{block}{Jawaban:}
    Berdasarkan **Teorema Keunikan (*Uniqueness Theorem*)** dan **Prinsip Ekstremum**:
    Karena nilai batas di semua sisi adalah $0\text{ V}$, maka nilai maksimum dan minimum keduanya adalah $0\text{ V}$.
    Maka nilai potensial di seluruh titik di dalam rongga adalah:
    $$ {\color{unibblue} V(x,y) = 0\text{ Volt}} $$
  \end{block}
\end{frame}

\begin{frame}{Rangkuman}
  \begin{enumerate}
    \item Persamaan Laplace adalah persamaan eliptik dasar untuk medan elektrostatik, magnetostatik, dan aliran panas steady-state.
    \item Fungsi solusi selalu bersifat halus (*smooth*) dan memenuhi sifat nilai rata-rata.
    \item Metode relaksasi Gauss-Seidel merupakan algoritma komputasi paling intuitif untuk menyelesaikan masalah elektrostatik 2D.
  \end{enumerate}
\end{frame}
