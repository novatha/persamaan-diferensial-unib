% FRAME 3: Peta Kurikulum & Koordinat Silinder
% -------------------------------------------------------------
\begin{frame}{Peta Kurikulum: Geometri Melingkar \& Koefisien Variabel}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Mengapa Koefisien Tidak Lagi Konstan?}
        \small
        \begin{itemize}
          \item Saluran kabel daya, serat optik, dan kawat berpenampang silinder memiliki simetri radial $(r, \phi, z)$.
          \item Faktor kelengkungan ruang memunculkan suku metrik $1/r$ dan $1/r^2$ pada Laplacian:
          \begin{equation*}
            \nabla^2 u = \frac{1}{r}\frac{\partial}{\partial r}\left(r\frac{\partial u}{\partial r}\right) + \frac{1}{r^2}\frac{\partial^2 u}{\partial \phi^2} + \frac{\partial^2 u}{\partial z^2}
          \end{equation*}
          \item PDB spasial arah radial berubah menjadi \textbf{PDB dengan koefisien variabel}!
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Solusi Fungsi Khusus Bessel}
        \small
        \begin{itemize}
          \item PDB koefisien variabel tidak dapat diselesaikan dengan akar karakteristik aljabar sederhana.
          \item Membutuhkan ekspansi deret pangkat khusus: \textbf{Metode Frobenius}.
          \item Menghasilkan fungsi basis ortogonal baru: \textbf{Fungsi Bessel}, yang setara dengan fungsi sinus/kosinus untuk geometri silinder!
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
  \vspace{-0.05cm}
  \begin{exampleblock}{Prinsip Fisis Efek Kulit (Skin Effect)}
    \footnotesize
    Pada frekuensi AC, fluks magnetik internal menginduksi ggl lawan yang mendesak arus listrik mengalir hanya di dekat permukaan luar konduktor.
  \end{exampleblock}
\end{frame}

% FRAME 4: Penurunan Persamaan Diferensial Bessel
% -------------------------------------------------------------
\begin{frame}{Penurunan Persamaan Diferensial Bessel}
  Tinjau persamaan gelombang Helmholtz silindris: $\nabla^2 u + k^2 u = 0$.
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Pemisahan Variabel Radial \& Azimuth}
        \small
        Asumsikan $u(r, \phi) = R(r) \cdot \Phi(\phi)$:\\
        $\frac{\Phi''}{\Phi} = -\nu^2 \implies \Phi(\phi) = e^{j\nu \phi}$ ($\nu \in \mathbb{Z}$).\\
        Substitusi ke komponen radial:
        \begin{equation*}
          \frac{1}{r}\frac{d}{dr}\left(r \frac{dR}{dr}\right) + \left(k^2 - \frac{\nu^2}{r^2}\right)R = 0
        \end{equation*}
        Kalikan seluruh persamaan dengan $r^2$:
        \begin{equation*}
          r^2 \frac{d^2 R}{dr^2} + r \frac{dR}{dr} + (k^2 r^2 - \nu^2)R = 0
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Bentuk Baku Persamaan Bessel}
        \small
        Substitusi variabel tak berdimensi $x = kr$:
        \begin{equation*}
          \mathbf{x^2 \frac{d^2 y}{dx^2} + x \frac{dy}{dx} + (x^2 - \nu^2)y = 0}
        \end{equation*}
        \textbf{Parameter Matematis:}
        \begin{itemize}
          \item Orde $\nu \ge 0$: Menentukan keteraturan fisis terhadap sudut putar azimuth $\phi$.
          \item Titik $x = 0$ ($r = 0$) adalah \textbf{Titik Singularitas Reguler}.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 5: Metode Deret Frobenius
% -------------------------------------------------------------
\begin{frame}{Metode Deret Frobenius di Sekitar Titik Singular Reguler}
  \begin{block}{Ekspansi Deret Frobenius di Sekitar $x = 0$}
    \small
    Karena titik $x = 0$ adalah singularitas reguler, solusi diasumsikan dalam bentuk deret tergeneralisasi:
    \begin{equation*}
      y(x) = x^r \sum_{m=0}^{\infty} c_m x^m = \sum_{m=0}^{\infty} c_m x^{m+r} \quad (c_0 \ne 0)
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Persamaan Indisial (\textit{Indicial Equation})}
        \small
        Substitusikan deret dan kumpulkan suku berpangkat terendah ($x^r$):
        \begin{equation*}
          r(r - 1) + r - \nu^2 = 0 \implies \mathbf{r^2 - \nu^2 = 0}
        \end{equation*}
        Akar persamaan indisial:
        \begin{equation*}
          \mathbf{r_1 = +\nu, \quad r_2 = -\nu}
        \end{equation*}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Hubungan Rekursi Koefisien}
        \small
        Penyamaan koefisien suku sejenis menghasilkan relasi:
        \begin{equation*}
          c_{2m} = \frac{(-1)^m}{2^{2m} m! (\nu + 1)(\nu + 2)\dots(\nu + m)} c_0
        \end{equation*}
        Seluruh suku bernomor ganjil lenyap identik ($c_1 = c_3 = \dots = 0$).
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 6: Fungsi Bessel Jenis Pertama & Jenis Kedua
% -------------------------------------------------------------
\begin{frame}{Fungsi Bessel Jenis Pertama $J_\nu(x)$ \& Jenis Kedua $Y_\nu(x)$}
  Solusi lengkap persamaan Bessel merupakan kombinasi linier dari dua fungsi bebas linier:
  \begin{equation*}
    \mathbf{y(x) = A J_\nu(x) + B Y_\nu(x)}
  \end{equation*}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Fungsi Bessel Jenis Pertama: $J_\nu(x)$}
        \small
        \textbf{Karakteristik Fisik:}
        \begin{itemize}
          \item Terhingga dan teratur di titik pusat $x = 0$: $J_0(0) = 1$, $J_\nu(0) = 0$ ($\nu \ge 1$).
          \item Menyerupai kosinus yang teredam asimtotik: $\approx \sqrt{\frac{2}{\pi x}}\cos(x - \frac{\nu\pi}{2} - \frac{\pi}{4})$.
          \item \textbf{Wajib Dipakai} pada domain yang mencakup sumbu tengah kawat pejal ($r = 0$).
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Fungsi Bessel Jenis Kedua: $Y_\nu(x)$}
        \small
        \textbf{Karakteristik Fisik:}
        \begin{itemize}
          \item \textbf{Meledak ke $-\infty$} di titik pusat: $\lim_{x \to 0} Y_\nu(x) = -\infty$.
          \item Sering disebut Fungsi Neumann $N_\nu(x)$.
          \item \textbf{Ditolak ($B = 0$)} pada konduktor pejal pejal karena potensial tidak boleh tak hingga di pusat kawat.
          \item \textbf{Dipakai} pada ruang cincin antar-konduktor (kabel koaksial $a < r < b$).
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 7: Akar-Akar Nol Bessel & Pandu Gelombang
% -------------------------------------------------------------
\begin{frame}{Akar-Akar Nol Bessel ($\alpha_{nm}$) \& Frekuensi Pancung}
  \begin{block}{Syarat Batas Dirichlet pada Dinding Silinder Logam}
    \small
    Pada tabung pemandu gelombang berjari-jari $a$, komponen medan tangensial harus lenyap: $E_z(a) = 0 \implies J_n(k_c a) = 0$.
  \end{block}
  \vspace{0.1cm}
  \begin{table}[h]
    \centering
    \scriptsize
    \begin{tabular}{lcccc}
      \toprule
      \textbf{Orde Bessel ($n$)} & \textbf{Akar ke-1 ($\alpha_{n,1}$)} & \textbf{Akar ke-2 ($\alpha_{n,2}$)} & \textbf{Akar ke-3 ($\alpha_{n,3}$)} & \textbf{Ragam Gelombang TM} \\
      \midrule
      $n = 0$ ($J_0(x) = 0$) & $\mathbf{2{,}4048}$ & $5{,}5201$ & $8{,}6537$ & Ragam $\text{TM}_{01}$ \\
      $n = 1$ ($J_1(x) = 0$) & $\mathbf{3{,}8317}$ & $7{,}0156$ & $10{,}1735$ & Ragam $\text{TM}_{11}$ \\
      $n = 2$ ($J_2(x) = 0$) & $\mathbf{5{,}1356}$ & $8{,}4172$ & $11{,}6198$ & Ragam $\text{TM}_{21}$ \\
      \bottomrule
    \end{tabular}
  \end{table}
  \begin{alertblock}{Frekuensi Pancung (\textit{Cut-off Frequency}) Ragam Silindris}
    \small
    \begin{equation*}
      \mathbf{f_{c,nm} = \frac{\alpha_{nm} \cdot c}{2\pi a}}
    \end{equation*}
    Sinyal microwave di bawah $f_c$ tidak dapat merambat dan akan padam teratenuasi eksponensial!
  \end{alertblock}
\end{frame}

\begin{frame}{Contoh Terhitung: Frekuensi Pancung Pandu Gelombang $\text{TM}_{01}$}
  \begin{exampleblock}{Masalah: Tabung pemandu gelombang silindris ($a = 2{,}5\text{ cm} = 0{,}025\text{ m}$)}
    Hitung bilangan gelombang pancung $k_c$ dan frekuensi pancung $f_c$ ragam $\text{TM}_{01}$ ($\alpha_{0,1} = 2{,}405$) di udara!
  \end{exampleblock}
  \vspace{-0.05cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penentuan Bilangan Gelombang $k_c$}
        \small
        Syarat batas dinding silinder:
        \begin{equation*}
          J_0(k_c a) = 0 \implies k_c a = \alpha_{0,1} = 2{,}405
        \end{equation*}
        Maka bilangan gelombang pancung:
        \begin{equation*}
          \mathbf{k_c = \frac{\alpha_{0,1}}{a} = \frac{2{,}405}{0{,}025} = 96{,}2\text{ rad/m}}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Perhitungan Frekuensi Pancung $f_c$}
        \small
        Hubungan dispersi gelombang EM ($c = 3 \times 10^8\text{ m/s}$):
        \begin{align*}
          f_c &= \frac{k_c \cdot c}{2\pi} = \frac{96{,}2 \times 3 \times 10^8}{2\pi} \\
              &= \frac{2{,}886 \times 10^{10}}{6{,}2832} \approx \mathbf{4{,}593\text{ GHz}} \quad \qed
        \end{align*}
        \textbf{Kesimpulan:} Sinyal microwave $f < 4{,}593\text{ GHz}$ terblokir total!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contoh Terhitung: Kedalaman Kulit Tembaga \& Aluminium}
  \begin{exampleblock}{Masalah: Hitung kedalaman penetrasi kulit ($\delta$) pada frekuensi $50\text{ Hz}$ dan $1\text{ MHz}$}
    Konduktivitas tembaga $\sigma_{\text{Cu}} = 5{,}8 \times 10^7\text{ S/m}$, aluminium $\sigma_{\text{Al}} = 3{,}5 \times 10^7\text{ S/m}$, $\mu = \mu_0 = 4\pi \times 10^{-7}\text{ H/m}$.
  \end{exampleblock}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Frekuensi Daya $50\text{ Hz}$}
        \small
        Rumus kedalaman kulit: $\delta = \frac{1}{\sqrt{\pi f \mu \sigma}}$.
        \begin{itemize}
          \item \textbf{Tembaga ($50\text{ Hz}$):}
          \begin{equation*}
            \delta_{\text{Cu}} = \frac{1}{\sqrt{\pi f \mu_0 \sigma_{\text{Cu}}}} \approx \mathbf{9{,}22\text{ mm}}
          \end{equation*}
          \item \textbf{Aluminium ($50\text{ Hz}$):}
          \begin{equation*}
            \delta_{\text{Al}} = \frac{1}{\sqrt{\pi f \mu_0 \sigma_{\text{Al}}}} \approx \mathbf{11{,}84\text{ mm}}
          \end{equation*}
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Frekuensi Tinggi $1\text{ MHz}$ ($10^6\text{ Hz}$)}
        \small
        Karena $\delta \propto 1/\sqrt{f}$:
        \begin{align*}
          \delta_{\text{Cu}}(1\text{ MHz}) &= \delta_{\text{Cu}}(50\text{ Hz}) \times \sqrt{\frac{50}{10^6}} \\
          &= 9{,}22\text{ mm} \times 0{,}007071 \\
          &\approx \mathbf{0{,}0652\text{ mm} = 65{,}2\,\mu\text{m}} \quad \qed
        \end{align*}
        Pada $1\text{ MHz}$, arus hanya mengalir pada lapisan setipis selaput ($65\,\mu\text{m}$)!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Peningkatan Resistansi Efektif AC ($R_{\text{ac}}/R_{\text{dc}}$)}
  Karena penampang efektif kawat menyusut menjadi lapisan cincin tipis $\delta$, resistansi AC melonjak:
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Formula Asimtotik Rasio Resistansi ($a/\delta > 4$)}
        \small
        \begin{equation*}
          \mathbf{\frac{R_{\text{ac}}}{R_{\text{dc}}} \approx \frac{a}{2\delta} + 0{,}25}
        \end{equation*}
        \textbf{Contoh Kawat $d = 2\text{ cm}$ ($a = 10\text{ mm}$) pada $1\text{ kHz}$ ($\delta = 2{,}06\text{ mm}$):}
        \begin{equation*}
          \frac{R_{\text{ac}}}{R_{\text{dc}}} = \frac{10}{2(2{,}06)} + 0{,}25 = 2{,}427 + 0{,}25 = \mathbf{2{,}68}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Dampak Lonjakan Rugi Daya ($I^2 R$)}
        \small
        Persentase kenaikan rugi-rugi panas Joule:
        \begin{equation*}
          \Delta P_{\%} = (2{,}68 - 1) \times 100\% = \mathbf{+168\%}
        \end{equation*}
        \textbf{Pemborosan Energi:}
        Bagian tengah konduktor pejal menjadi mubazir karena sama sekali tidak dilalui arus listrik, namun tetap menambah berat mekanis tarikan tiang saluran transmisi!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
