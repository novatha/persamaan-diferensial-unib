\section{Review: Persamaan Maxwell}
\begin{frame}{Persamaan Maxwell dalam Bentuk Diferensial}
  \begin{block}{Empat Persamaan Fundamental}
    Persamaan Maxwell dalam ruang hampa (tanpa muatan dan arus bebas):
    \begin{align}
        \nabla \cdot \mathbf{E} &= 0 \quad \text{(Hukum Gauss untuk Listrik)} \\
        \nabla \cdot \mathbf{B} &= 0 \quad \text{(Hukum Gauss untuk Magnet)} \\
        \nabla \times \mathbf{E} &= -\frac{\partial \mathbf{B}}{\partial t} \quad \text{(Hukum Faraday)} \\
        \nabla \times \mathbf{B} &= \mu_0 \epsilon_0 \frac{\partial \mathbf{E}}{\partial t} \quad \text{(Hukum Ampere-Maxwell)}
    \end{align}
  \end{block}
  \begin{itemize}
      \item Keempat persamaan ini merupakan \textbf{Persamaan Diferensial Parsial (PDP)} yang saling terikat (coupled).
  \end{itemize}
\end{frame}

\section{Kaitan PDP dan Gelombang EM}
\begin{frame}{Menurunkan Persamaan Gelombang (Uncoupled PDP)}
    Dengan menggunakan identitas vektor $\nabla \times (\nabla \times \mathbf{A}) = \nabla(\nabla \cdot \mathbf{A}) - \nabla^2 \mathbf{A}$, kita dapat memisahkan persamaan untuk $\mathbf{E}$:
    \begin{alertblock}{Penurunan untuk Medan Listrik}
    \begin{align*}
        \nabla \times (\nabla \times \mathbf{E}) &= \nabla \times \left(-\frac{\partial \mathbf{B}}{\partial t}\right) \\
        \nabla(\nabla \cdot \mathbf{E}) - \nabla^2 \mathbf{E} &= -\frac{\partial}{\partial t}(\nabla \times \mathbf{B}) \\
        0 - \nabla^2 \mathbf{E} &= -\frac{\partial}{\partial t} \left(\mu_0 \epsilon_0 \frac{\partial \mathbf{E}}{\partial t}\right) \\
        \nabla^2 \mathbf{E} &= \mu_0 \epsilon_0 \frac{\partial^2 \mathbf{E}}{\partial t^2}
    \end{align*}
    \end{alertblock}
    Persamaan terakhir adalah PDP Gelombang 3 Dimensi klasik!
\end{frame}
