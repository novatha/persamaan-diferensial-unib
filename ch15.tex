\begin{frame}{Persamaan Laplace Koordinat Bola \& Persamaan Legendre}
  Laplacian dalam koordinat bola simetri azimut ($\partial/\partial \phi = 0$):
  \begin{equation*}
    \nabla^2 V = \frac{1}{r^2}\frac{\partial}{\partial r}\left(r^2 \frac{\partial V}{\partial r}\right) + \frac{1}{r^2 \sin\theta}\frac{\partial}{\partial \theta}\left(\sin\theta \frac{\partial V}{\partial \theta}\right) = 0
  \end{equation*}
  \vspace{-0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Pemisahan Variabel $V(r, \theta) = R(r)\Theta(\theta)$}
        \small
        Substitusi menghasilkan konstanta pemisah $n(n+1)$:\\
        1. Persamaan Radial:
        \begin{equation*}
          r^2 R'' + 2r R' - n(n+1)R = 0
        \end{equation*}
        Solusi: $R(r) = A_n r^n + B_n r^{-(n+1)}$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Persamaan Diferensial Legendre}
        \small
        Dengan substitusi $x = \cos\theta$ (maka $dx = -\sin\theta\,d\theta$):
        \begin{equation*}
          \mathbf{(1 - x^2)\frac{d^2 \Theta}{dx^2} - 2x \frac{d\Theta}{dx} + n(n+1)\Theta = 0}
        \end{equation*}
        Solusi yang berhingga pada kutub ($x = \pm 1$) adalah \textbf{Polinomial Legendre} $P_n(x)$!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 5: Polinomial Legendre P_n(x) & Sifat Ortogonalitas
% -------------------------------------------------------------
\begin{frame}{Polinomial Legendre $P_n(\cos\theta)$ \& Sifat Ortogonalitas}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Formula Rodrigues ($n \in \mathbb{Z}^+$)}
        \small
        \begin{equation*}
          P_n(x) = \frac{1}{2^n n!} \frac{d^n}{dx^n}\left[(x^2 - 1)^n\right]
        \end{equation*}
        Empat orde pertama ($x = \cos\theta$):
        \begin{itemize}
          \item $P_0(x) = 1$ (Monopol)
          \item $P_1(x) = x = \cos\theta$ (Dipol)
          \item $P_2(x) = \frac{1}{2}(3x^2 - 1) = \frac{1}{2}(3\cos^2\theta - 1)$
          \item $P_3(x) = \frac{1}{2}(5x^3 - 3x)$ (Oktopol)
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Teorema Ortogonalitas}
        \small
        Polinomial Legendre saling ortogonal pada interval $[-1, 1]$:
        \begin{equation*}
          \int_{-1}^{1} P_m(x) P_n(x)\,dx = \frac{2}{2n + 1} \delta_{mn}
        \end{equation*}
        \textbf{Solusi Umum Potensial Bola:}
        \begin{equation*}
          V(r, \theta) = \sum_{n=0}^{\infty} \left(A_n r^n + \frac{B_n}{r^{n+1}}\right) P_n(\cos\theta)
        \end{equation*}
        Digunakan untuk ekspansi multipol medan elektrostatika.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 6: Aplikasi: Bola Konduktor dalam Medan Seragam

\begin{frame}{Aplikasi: Bola Konduktor dalam Medan Seragam $E_0 \hat{a}_z$}
  \begin{exampleblock}{Masalah Nilai Batas Elektrostatik}
    Bola konduktor netral berjari-jari $a$ diletakkan dalam medan luar seragam $\vec{E} = E_0 \hat{a}_z$. Tentukan $V(r, \theta)$!
  \end{exampleblock}
  \vspace{-0.05cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Syarat Batas Fisis}
        \small
        \begin{enumerate}
          \item $V \to -E_0 z = -E_0 r\cos\theta$ saat $r \to \infty$ $\implies A_1 = -E_0$, $A_{n \ne 1} = 0$.
          \item Permukaan bola ekuipotensial:\\
          $V(a, \theta) = 0 \implies -E_0 a + \frac{B_1}{a^2} = 0 \implies B_1 = E_0 a^3$.
        \end{enumerate}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Solusi Eksak Superposisi}
        \small
        \begin{equation*}
          \mathbf{V(r, \theta) = -E_0 r\cos\theta + \frac{E_0 a^3}{r^2}\cos\theta}
        \end{equation*}
        \begin{itemize}
          \item Suku 1: Medan luar tak terganggu.
          \item Suku 2: Medan dipol induksi muatan terdistribusi pada permukaan bola!
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Empat Persamaan Diferensial Maxwell}
  Hukum fundamental kelistrikan dan kemagnetan dalam bentuk diferensial:
  \vspace{0.05cm}
  \begin{table}[h]
    \centering
    \resizebox{\linewidth}{!}{%
    \begin{tabular}{lll}
      \toprule
      \textbf{Hukum Fisika} & \textbf{Bentuk Diferensial} & \textbf{Makna Fisika Rekayasa Elektro} \\
      \midrule
      Hukum Gauss Listrik & $\nabla \cdot \vec{D} = \rho_v$ & Fluks listrik bersumber dari muatan bebas volume. \\
      Hukum Gauss Magnet & $\nabla \cdot \vec{B} = 0$ & Ketiadaan monopol magnet; garis fluks selalu tertutup. \\
      Hukum Faraday & $\nabla \times \vec{E} = -\frac{\partial \vec{B}}{\partial t}$ & Medan magnetik berubah waktu menimbulkan vorteks medan listrik (GGL). \\
      Hukum Ampere-Maxwell & $\nabla \times \vec{H} = \vec{J} + \frac{\partial \vec{D}}{\partial t}$ & Arus konduksi \& \textbf{arus pergeseran} menimbulkan vorteks medan magnet. \\
      \bottomrule
    \end{tabular}%
    }
  \end{table}
  \vspace{-0.1cm}
  \begin{block}{Hubungan Konstitutif Material Linear Isotropis}
    \centering
    $\vec{D} = \epsilon \vec{E} = \epsilon_r \epsilon_0 \vec{E}, \quad \vec{B} = \mu \vec{H} = \mu_r \mu_0 \vec{H}, \quad \vec{J} = \sigma \vec{E}$ (Hukum Ohm Titik)
  \end{block}
\end{frame}

% FRAME 8: Arus Pergeseran Maxwell (Displacement Current)
% -------------------------------------------------------------
\begin{frame}{Arus Pergeseran Maxwell (\textit{Displacement Current})}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Inkonsistensi Matematis Hukum Ampere Lama}
        \small
        Hukum Ampere statik: $\nabla \times \vec{H} = \vec{J}$.\\
        Terapkan divergensi pada kedua ruas:
        \begin{equation*}
          \nabla \cdot (\nabla \times \vec{H}) \equiv 0 \implies \nabla \cdot \vec{J} = 0
        \end{equation*}
        Namun hukum kekekalan muatan menyatakan persamaan kontinuitas:
        \begin{equation*}
          \mathbf{\nabla \cdot \vec{J} = -\frac{\partial \rho_v}{\partial t} \ne 0 \quad (\text{Kontradiksi!})}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Koreksi Cemerlang Maxwell}
        \small
        Substitusi Hukum Gauss $\rho_v = \nabla \cdot \vec{D}$:
        \begin{equation*}
          \nabla \cdot \vec{J} + \frac{\partial}{\partial t}(\nabla \cdot \vec{D}) = 0 \implies \nabla \cdot \left(\vec{J} + \frac{\partial \vec{D}}{\partial t}\right) = 0
        \end{equation*}
        Maxwell menambahkan suku \textbf{Rapat Arus Pergeseran}:
        \begin{equation*}
          \mathbf{\vec{J}_d = \frac{\partial \vec{D}}{\partial t} = \epsilon \frac{\partial \vec{E}}{\partial t}}
        \end{equation*}
        Medan listrik berubah waktu berlaku layaknya arus fisis!
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Contoh Terhitung: Kesamaan $I_c$ \& $I_d$ pada Kapasitor}
  \begin{exampleblock}{Masalah: Kapasitor pelat sejajar luas $A$, jarak $d$, dielektrik $\epsilon$, diberi tegangan $v(t) = V_m \sin(\omega t)$}
    Buktikan bahwa arus konduksi $i_c(t)$ pada kawat sama persis dengan arus pergeseran $i_d(t)$ di celah dielektrik!
  \end{exampleblock}
  \vspace{-0.05cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{1. Arus Konduksi pada Kawat Luar}
        \small
        Kapasitansi pelat: $C = \frac{\epsilon A}{d}$.\\
        Arus konduksi pengisian kapasitor:
        \begin{align*}
          i_c(t) &= C \frac{dv}{dt} = \left(\frac{\epsilon A}{d}\right) \frac{d}{dt}\left[V_m \sin(\omega t)\right] \\
          &= \mathbf{\frac{\epsilon A \omega V_m}{d} \cos(\omega t)}
        \end{align*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{2. Arus Pergeseran di Celah Dielektrik}
        \small
        Medan listrik celah: $E(t) = \frac{v(t)}{d} = \frac{V_m}{d}\sin(\omega t)$.\\
        Rapat fluks: $D(t) = \epsilon E(t) = \frac{\epsilon V_m}{d}\sin(\omega t)$.\\
        Integralkan rapat arus pergeseran pada luas $A$:
        \begin{align*}
          i_d(t) &= \int_A \frac{\partial D}{\partial t} dA = A \cdot \frac{\partial D}{\partial t} \\
          &= \mathbf{\frac{\epsilon A \omega V_m}{d} \cos(\omega t) \equiv i_c(t)} \quad \qed
        \end{align*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 10: Penurunan Persamaan Gelombang Elektromagnetik

\begin{frame}{Penurunan Persamaan Gelombang Elektromagnetik 3D}
  Tinjau ruang hampa tanpa muatan ($\rho_v = 0$) dan tanpa konduksi ($\vec{J} = 0$):
  \vspace{-0.1cm}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Operasi Curl terhadap Hukum Faraday}
        \small
        \begin{equation*}
          \nabla \times (\nabla \times \vec{E}) = \nabla \times \left(-\mu_0 \frac{\partial \vec{H}}{\partial t}\right) = -\mu_0 \frac{\partial}{\partial t}(\nabla \times \vec{H})
        \end{equation*}
        Substitusi Ampere-Maxwell ($\nabla \times \vec{H} = \epsilon_0 \frac{\partial \vec{E}}{\partial t}$):
        \begin{equation*}
          \nabla \times (\nabla \times \vec{E}) = -\mu_0 \epsilon_0 \frac{\partial^2 \vec{E}}{\partial t^2}
        \end{equation*}
        Identitas Vektor Kalkulus:
        \begin{equation*}
          \nabla \times (\nabla \times \vec{E}) = \nabla(\underbrace{\nabla \cdot \vec{E}}_{=0}) - \nabla^2 \vec{E} = -\nabla^2 \vec{E}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Persamaan Gelombang 3D \& Kecepatan Cahaya}
        \small
        Persamaan Diferensial Parsial Hiperbolik:
        \begin{equation*}
          \mathbf{\nabla^2 \vec{E} - \mu_0 \epsilon_0 \frac{\partial^2 \vec{E}}{\partial t^2} = 0}
        \end{equation*}
        Bandingkan dengan gelombang standar $\nabla^2 u - \frac{1}{v^2}\frac{\partial^2 u}{\partial t^2} = 0$:
        \begin{align*}
          c &= \frac{1}{\sqrt{\mu_0 \epsilon_0}} = \frac{1}{\sqrt{(4\pi \cdot 10^{-7})(8{,}854 \cdot 10^{-12})}} \\
            &\approx \mathbf{2{,}998 \times 10^8\text{ m/s}} \quad (\text{Kecepatan Cahaya!})
        \end{align*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Aliran Energi Gelombang \& Vektor Poynting}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Definisi Vektor Poynting $\vec{S}$}
        \small
        Laju perpindahan energi elektromagnetik per satuan luas ($\text{W/m}^2$):
        \begin{equation*}
          \mathbf{\vec{S} = \vec{E} \times \vec{H}}
        \end{equation*}
        Untuk gelombang datar harmonik waktu:
        \begin{equation*}
          \langle \vec{S} \rangle = \frac{1}{2} \text{Re}\{\vec{E} \times \vec{H}^*\} = \frac{|\vec{E}|^2}{2\eta} \hat{a}_k
        \end{equation*}
        Arah rambat energi tegak lurus terhadap bidang $\vec{E}$ dan $\vec{H}$.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Impedansi Intrinsik Medium $\eta$}
        \small
        Rasio amplitudo medan listrik terhadap magnet:
        \begin{equation*}
          \mathbf{\eta = \frac{|\vec{E}|}{|\vec{H}|} = \sqrt{\frac{\mu}{\epsilon}}}
        \end{equation*}
        Pada ruang bebas / udara ($\mu_0, \epsilon_0$):
        \begin{equation*}
          \mathbf{\eta_0 = \sqrt{\frac{\mu_0}{\epsilon_0}} \approx 120\pi \approx 376{,}73\,\Omega \approx 377\,\Omega}
        \end{equation*}
        Konstanta fundamental propagasi gelombang elektromagnetik.
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
