% FRAME 4: Transien RL & Inductive Kickback
\begin{frame}{Transien Pemutusan Sirkuit RL: Lonjakan Tegangan Balik}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Fenomena \textit{Inductive Kickback}:}
    \begin{itemize}
      \item Pada kondisi tunak DC, arus koil mengalir mantap: $I_0 = \frac{V_0}{R}$.
      \item Bila sakelar tiba-tiba dibuka pada $t=0$, arus dipaksa turun ke nol dalam waktu sangat singkat $\Delta t \to 0$:
      \begin{equation*}
        \left| \frac{di}{dt} \right| \to \infty
      \end{equation*}
      \item Induktor merespons dengan membangkitkan tegangan lawan (\textit{back-EMF}):
      \begin{equation*}
        \mathbf{v_L(t) = L \frac{di}{dt} \gg V_0}
      \end{equation*}
      \item Tegangan balik ini dapat mencapai \textbf{ribuan Volt}, memicu busur api atau merusak transistor!
    \end{itemize}

    \column{0.44\textwidth}
    \begin{alertblock}{Bahaya Kerusakan Semikonduktor}
      Transistor sakelar DC (MOSFET/BJT) memiliki batas tegangan maksimum $V_{\text{DS, max}}$. Lonjakan induktif yang tidak diredam akan menembus batas ini dalam orde mikrodetik.
    \end{alertblock}
  \end{columns}
\end{frame}

    \end{center}
    \vspace{-0.2cm}
    \begin{center}
      \footnotesize Rangkaian Proteksi Dioda Freewheeling Paralel Beban Induktif.
    \end{center}

    \column{0.54\textwidth}
    \textbf{Mekanisme Pelepasan Energi Sirkuit:}
    \begin{itemize}
      \item Saat sakelar tertutup, dioda terbias mundur (\textit{reverse biased}) dan tidak aktif.
      \item Saat sakelar dibuka, induktor membalik polaritas tegangannya, memicu dioda terbias maju (\textit{forward biased}).
      \item Arus induktor bersirkulasi aman melalui loop koil-dioda:
      \begin{equation*}
        L \frac{di}{dt} + R i + V_F = 0 \implies i(t) \approx I_0 e^{-\frac{t}{\tau}}
      \end{equation*}
      \item Tegangan pada sakelar dibatasi hanya sebesar $\mathbf{V_{\text{sakelar}} = V_0 + V_F \approx V_0 + 0{,}7\text{ V}}$.
    \end{itemize}
  \end{columns}
\end{frame}

% FRAME 6: Transien RC Pensaklaran Multi-Interval
\begin{frame}{Transien Pensaklaran Multi-Interval Sirkuit RC}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Prinsip Pergeseran Waktu Acuan:}
    \begin{itemize}
      \item Pada sirkuit dengan pensaklaran bertahap (misal $t_0 = 0$ dan $t_1 = t_s$):
      \item \textbf{Interval 1 ($0 \le t < t_s$):}
      \begin{equation*}
        v_C(t) = V_{\text{target},1} + [v_C(0) - V_{\text{target},1}] e^{-t/\tau_1}
      \end{equation*}
      \item Evaluasi nilai batas: $v_C(t_s^-) = v_C(t_s^+)$ (\textbf{Prinsip Kontinuitas}).
      \item \textbf{Interval 2 ($t \ge t_s$):}
      \begin{equation*}
        v_C(t) = V_{\text{target},2} + [v_C(t_s) - V_{\text{target},2}] e^{-\frac{t - t_s}{\tau_2}}
      \end{equation*}
    \end{itemize}

    \column{0.44\textwidth}

% FRAME 7: Hukum Pendinginan Newton dalam Sistem Tenaga
\begin{frame}{Dinamika Termal Sistem Tenaga: Hukum Pendinginan Newton}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Kesetimbangan Termal Transformator Daya:}
    \begin{itemize}
      \item Laju akumulasi panas di dalam tangki trafo sama dengan total daya rugi ($P_{\text{loss}} = I^2 R + P_{\text{core}}$) dikurangi laju pelepasan kalor ke udara ambien:
      \begin{equation}
        q_{\text{akumulasi}} = q_{\text{masuk}} - q_{\text{lepas}}
      \end{equation}
      \item Akumulasi panas: $q_{\text{akumulasi}} = C_{\text{th}} \frac{dT(t)}{dt}$
      \item Pelepasan kalor konveksi: $q_{\text{lepas}} = \frac{T(t) - T_{\text{amb}}}{R_{\text{th}}}$
    \end{itemize}

    \column{0.44\textwidth}
    \begin{block}{PDB Termal Orde 1 Standar Trafo}
      \begin{equation}
        C_{\text{th}} \frac{dT}{dt} + \frac{1}{R_{\text{th}}} (T - T_{\text{amb}}) = P_{\text{loss}}
      \end{equation}
      Solusi kenaikan suhu transien:
      \begin{equation*}
        \mathbf{T(t) = T_{\text{amb}} + P_{\text{loss}} R_{\text{th}} \left(1 - e^{-\frac{t}{\tau_{\text{th}}}}\right)}
      \end{equation*}
      dengan $\mathbf{\tau_{\text{th}} = R_{\text{th}} C_{\text{th}}}$ [detik atau jam].
    \end{block}
  \end{columns}
\end{frame}

% FRAME 9: Standar Penuaan Insulasi Trafo (IEEE Std C57.91)
\begin{frame}{Standar Industri: Penuaan Insulasi Trafo (IEEE Std C57.91)}
  \begin{columns}[c]
    \column{0.55\textwidth}
    \textbf{Hukum Penuaan Kimiawi Kertas Isolasi (Arrhenius):}
    \begin{itemize}
      \item Umur pakai transformator ditentukan oleh degradasi kertas isolasi belitan.
      \item Faktor laju penuaan relatif (\textit{Aging Acceleration Factor} / $F_{AA}$):
      \begin{equation*}
        \mathbf{F_{AA} = \exp\left[ \frac{15000}{383} - \frac{15000}{\Theta_H + 273} \right]}
      \end{equation*}
      di mana $\Theta_H$ adalah temperatur titik terpanas (\textit{hotspot temperature}) [$^\circ$C].
      \item \textbf{Aturan Praktis}: Kenaikan suhu hotspot sebesar $+6^\circ\text{C}$ menggandakan laju degradasi isolasi ($2\times$ lebih cepat aus).
    \end{itemize}

    \column{0.45\textwidth}
    \begin{alertblock}{Batas Operasi Gardu Induk PLN}
      \begin{itemize}
        \item Suhu Ambien Nominal: $30^\circ\text{C}$
        \item Batas Alarm Minyak Atas: $85^\circ\text{C}$
        \item Batas Trip Pemutus (CB): $95^\circ\text{C}$
        \item Batas Hotspot Darurat: $110^\circ\text{C}$ ($F_{AA} \approx 1{,}0$)
      \end{itemize}
    \end{alertblock}
  \end{columns}
\end{frame}

% FRAME 10: Worked Example 1 - Inductive Kickback
\begin{frame}{Contoh Soal 1 (Worked Example): Inductive Kickback Koil}
  \textbf{Kasus Industri:} Koil kontaktor magnetik panel DC memiliki $L = 2{,}5\text{ H}$ dan $R = 50\,\Omega$, dicatu $V_0 = 24\text{ V}$. Sakelar dibuka mendadak dan busur api memutus arus dalam $\Delta t = 2\,\mu\text{s}$.
  \vspace{0.15cm}

  \begin{columns}[t]
    \column{0.5\textwidth}
    \textbf{1. Arus Tunak Awal \& Konstanta Waktu:}
    \begin{align*}
      I_0 &= \frac{V_0}{R} = \frac{24\text{ V}}{50\,\Omega} = \mathbf{0{,}48\text{ A}} \\
      \tau &= \frac{L}{R} = \frac{2{,}5\text{ H}}{50\,\Omega} = \mathbf{50\text{ ms}}
    \end{align*}
    \textbf{2. Lonjakan Tegangan Balik Tanpa Dioda:}
    \begin{equation*}
      |v_L| = L \left| \frac{\Delta i}{\Delta t} \right| = 2{,}5 \times \frac{0{,}48}{2 \times 10^{-6}} = \mathbf{600.000\text{ Volt!}}
    \end{equation*}

    \column{0.5\textwidth}
    \textbf{3. Tegangan Sakelar dengan Dioda Flyback:}
    Jika dipasang dioda silikon freewheeling ($V_F = 0{,}7\text{ V}$):
    \begin{equation*}
      V_{\text{sakelar, max}} = V_0 + V_F = 24 + 0{,}7 = \mathbf{24{,}7\text{ V}}
    \end{equation*}
    Transistor MOSFET ($V_{\text{rated}} = 100\text{ V}$) \textbf{aman total}!
    \vspace{0.15cm}

    \textbf{4. Energi Medan Magnet yang Dilepas:}
    \begin{equation*}
      E_L = \frac{1}{2} L I_0^2 = \frac{1}{2}(2{,}5)(0{,}48)^2 = \mathbf{0{,}288\text{ Joule}}
    \end{equation*}
  \end{columns}
\end{frame}

% FRAME 11: Worked Example 2 - Dinamika Termal Trafo
\begin{frame}{Contoh Soal 2 (Worked Example): Manajemen Termal Trafo 60 MVA}
  \textbf{Kasus Gardu Induk PLN:} Trafo 60 MVA memiliki $C_{\text{th}} = 3{,}6 \times 10^6\text{ J}/^\circ\text{C}$, $R_{\text{th}} = 1{,}25 \times 10^{-4}\;^\circ\text{C}/\text{W}$. Suhu ambien $T_{\text{amb}} = 30^\circ\text{C}$, suhu awal $T_0 = 55^\circ\text{C}$. Lonjakan beban puncak menimbulkan rugi-rugi $P_{\text{loss}} = 480\text{ kW}$.
  \vspace{0.15cm}

  \begin{columns}[t]
    \column{0.5\textwidth}
    \textbf{1. Konstanta Waktu Termal Trafo:}
    \begin{align*}
      \tau_{\text{th}} &= R_{\text{th}} C_{\text{th}} = (1{,}25 \times 10^{-4})(3{,}6 \times 10^6) \\
                       &= 450\text{ detik} = \mathbf{7{,}5\text{ menit}}
    \end{align*}
    \textbf{2. Temperatur Tunak Akhir ($t \to \infty$):}
    \begin{align*}
      T_{\text{ss}} &= T_{\text{amb}} + P_{\text{loss}} R_{\text{th}} \\
                    &= 30^\circ\text{C} + (480.000)(1{,}25 \times 10^{-4}) = \mathbf{90^\circ\text{C}}
    \end{align*}
    Trafo melebihi alarm ($85^\circ\text{C}$) namun di bawah trip ($95^\circ\text{C}$).

    \column{0.5\textwidth}
    \textbf{3. Waktu Menuju Batas Alarm ($T = 85^\circ\text{C}$):}
    \begin{equation*}
      T(t) = 90 - (90 - 55) e^{-t/\tau_{\text{th}}} = 90 - 35 e^{-t/7{,}5}
    \end{equation*}
    Masukkan batas alarm $T = 85^\circ\text{C}$:
    \begin{align*}
      85 &= 90 - 35 e^{-t/7{,}5} \implies 35 e^{-t/7{,}5} = 5 \\
      e^{-t/7{,}5} &= \frac{5}{35} = \frac{1}{7} \implies \frac{t}{7{,}5} = \ln(7) = 1{,}946 \\
      t_{\text{alarm}} &= 7{,}5 \times 1{,}946 = \mathbf{14{,}6\text{ menit}}
    \end{align*}
    Operator memiliki waktu \textbf{14,6 menit} untuk manuver beban!
  \end{columns}
\end{frame}
