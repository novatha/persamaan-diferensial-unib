% FRAME 4: Bentuk Standar & Persamaan Karakteristik
\begin{frame}{Bentuk Standar PDB Orde 2 Homogen \& Persamaan Karakteristik}
  \begin{block}{Bentuk Umum PDB Linier Orde 2 Homogen}
    \begin{equation}
      a \frac{d^2 y}{dx^2} + b \frac{dy}{dx} + c y = 0, \quad (a \neq 0)
    \end{equation}
  \end{block}
  \textbf{Postulat Solusi Basis Eksponensial:}
  Asumsikan solusi berbentuk $y(x) = e^{rx}$, maka $y'(x) = r e^{rx}$ dan $y''(x) = r^2 e^{rx}$. Substitusikan:
  \begin{equation*}
    (a r^2 + b r + c) e^{rx} = 0 \implies \mathbf{a r^2 + b r + c = 0} \quad \text{(Persamaan Karakteristik)}
  \end{equation*}
  Akar-akar kuadrat ditentukan oleh rumus kuadratik abc:
  \begin{equation}
    r_{1,2} = \frac{-b \pm \sqrt{b^2 - 4ac}}{2a}, \quad \text{dengan Diskriminan } \mathbf{D = b^2 - 4ac}
  \end{equation}
\end{frame}

% FRAME 5: Teorema Superposisi & Determinan Wronskian
\begin{frame}{Teorema Superposisi \& Determinan Wronskian}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \begin{block}{Teorema Superposisi Linier}
      Jika $y_1(x)$ dan $y_2(x)$ adalah dua solusi bebas linier dari persamaan homogen, maka kombinasi linier keduanya:
      \begin{equation*}
        \mathbf{y(x) = C_1 y_1(x) + C_2 y_2(x)}
      \end{equation*}
      adalah solusi umum lengkap.
    \end{block}

    \column{0.44\textwidth}
    \begin{alertblock}{Uji Bebas Linier Wronskian}
      Solusi $y_1$ dan $y_2$ saling bebas linier jika determinan Wronskian tidak nol:
      \begin{equation*}
        W(y_1, y_2) = \begin{vmatrix} y_1 & y_2 \\ y_1' & y_2' \end{vmatrix} = y_1 y_2' - y_1' y_2 \neq 0
      \end{equation*}
      Jika $W = 0$, solusi saling bergantung secara linier.
    \end{alertblock}
  \end{columns}
\end{frame}

% FRAME 6: Pemodelan Sirkuit RLC Seri
\begin{frame}{Pemodelan Fisis: Rangkaian RLC Seri Bebas Sumber}
  \begin{columns}
    \column{0.45\textwidth}
    \begin{center}
      \resizebox{\linewidth}{!}{%
      \begin{circuitikz}[american, scale=1.1]
        \draw (0,0) to[R, l=$R$, v=$v_R$] (0,2.5)
              to[L, l=$L$, v=$v_L$] (3.2,2.5)
              to[C, l=$C$, v=$v_C$] (3.2,0)
              -- (0,0);
        \draw[->, >=stealth, thick, unibblue] (1.6,1.2) arc (160:-160:0.6);
        \node at (2.2,1.2) [unibblue, font=\small\bfseries] {$i(t)$};
      \end{circuitikz}%
      }
    \end{center}
    \vspace{-0.2cm}
    \begin{center}
      \footnotesize Rangkaian RLC Seri Tertutup Bebas Sumber.
    \end{center}

    \column{0.55\textwidth}
    \textbf{Formulasi Hukum Tegangan Kirchhoff (KVL):}
    \begin{equation}
      v_R(t) + v_L(t) + v_C(t) = 0
    \end{equation}
    Nyatakan seluruh variabel dalam muatan kapasitor $q(t)$ ($i = \frac{dq}{dt}$, $v_C = \frac{q}{C}$):
    \begin{equation*}
      L \frac{d^2 q}{dt^2} + R \frac{dq}{dt} + \frac{1}{C} q = 0
    \end{equation*}
    Diferensiasikan terhadap waktu untuk memperoleh dinamika arus $i(t)$:
    \begin{block}{PDB Standar Rangkaian RLC Seri}
      \begin{equation}
        \frac{d^2 i(t)}{dt^2} + \frac{R}{L} \frac{di(t)}{dt} + \frac{1}{LC} i(t) = 0
      \end{equation}
    \end{block}
  \end{columns}
\end{frame}

% FRAME 7: Parameter Karakteristik RLC
\begin{frame}{Parameter Standar Karakteristik RLC}
  Dalam rekayasa elektro, PDB orde 2 dinyatakan dalam dua parameter kunci:
  \vspace{0.1cm}
  \begin{columns}[t]
    \column{0.5\textwidth}
    \begin{block}{Parameter Sistem}
      \begin{itemize}
        \item \textbf{Frekuensi Sudut Alami:}
        \begin{equation*}
          \omega_0 = \frac{1}{\sqrt{LC}} \quad [\text{rad/s}]
        \end{equation*}
        \item \textbf{Faktor Redaman Neper:}
        \begin{equation*}
          \alpha = \frac{R}{2L} \quad [\text{Np/s}]
        \end{equation*}
        \item \textbf{Rasio Redaman (\textit{Damping Ratio}):}
        \begin{equation*}
          \zeta = \frac{\alpha}{\omega_0} = \frac{R}{2} \sqrt{\frac{C}{L}}
        \end{equation*}
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{exampleblock}{Persamaan Karakteristik Rekayasa}
      \begin{equation*}
        s^2 + 2\alpha s + \omega_0^2 = 0 \iff s^2 + 2\zeta\omega_0 s + \omega_0^2 = 0
      \end{equation*}
      Akar-akar karakteristik:
      \begin{equation*}
        \mathbf{s_{1,2} = -\alpha \pm \sqrt{\alpha^2 - \omega_0^2}}
      \end{equation*}
      Nilai relatif $\alpha$ terhadap $\omega_0$ (atau $\zeta$ terhadap 1) memunculkan \textbf{Tiga Ragam Redaman}.
    \end{exampleblock}
  \end{columns}
\end{frame}

% FRAME 8: Ragam 1 - Overdamped
\begin{frame}{Ragam 1: Teredam Lebih (\textit{Overdamped}, $\alpha > \omega_0$ atau $\zeta > 1$)}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Karakteristik Fisik:}
    \begin{itemize}
      \item Terjadi jika $R > 2\sqrt{L/C}$ (resistansi peredam dominan).
      \item Diskriminan positif: $D > 0 \implies$ dua akar real negatif berbeda:
      \begin{equation*}
        s_1 = -\alpha + \sqrt{\alpha^2 - \omega_0^2}, \quad s_2 = -\alpha - \sqrt{\alpha^2 - \omega_0^2}
      \end{equation*}
      \item \textbf{Bentuk Solusi Umum:}
      \begin{equation*}
        \mathbf{i(t) = C_1 e^{s_1 t} + C_2 e^{s_2 t}}
      \end{equation*}
      \item Respon waktu murni peluruhan eksponensial lambat tanpa osilasi sama sekali.
    \end{itemize}

    \column{0.44\textwidth}
    \begin{center}
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[scale=0.85]
        \draw[->] (0,0) -- (4.5,0) node[right] {$t$};
        \draw[->] (0,0) -- (0,3) node[above] {$i(t)$};
        \draw[domain=0:4.2, smooth, blue!85!black, very thick] plot ({\x}, {3.5*(exp(-0.8*\x) - exp(-2.5*\x))});
        \node[blue!85!black, font=\scriptsize\bfseries] at (2.5, 1.2) {Overdamped};
      \end{tikzpicture}%
      }
    \end{center}
  \end{columns}
\end{frame}

% FRAME 9: Ragam 2 - Critically Damped
\begin{frame}{Ragam 2: Teredam Kritis (\textit{Critically Damped}, $\alpha = \omega_0$ atau $\zeta = 1$)}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Karakteristik Fisik:}
    \begin{itemize}
      \item Terjadi tepat saat $R = R_{\text{kritis}} = 2\sqrt{L/C}$.
      \item Diskriminan nol: $D = 0 \implies$ akar kembar negatif:
      \begin{equation*}
        s_1 = s_2 = -\alpha
      \end{equation*}
      \item Berdasarkan reduksi orde d'Alembert, solusi basis kedua dikalikan dengan $t$:
      \begin{equation*}
        \mathbf{i(t) = (C_1 + C_2 t) e^{-\alpha t}}
      \end{equation*}
      \item \textbf{Keistimewaan Rekayasa}: Merupakan respon tercepat mencapai kondisi tunak tanpa osilasi/overshoot.
    \end{itemize}

    \column{0.44\textwidth}
    \begin{alertblock}{Aplikasi Kritis}
      Diterapkan pada jarum ukur analog, peredam kejut mekanis, dan kontroler servo agar sistem segera tenang tanpa getaran bolak-balik.
    \end{alertblock}
  \end{columns}
\end{frame}

% FRAME 10: Ragam 3 - Underdamped
\begin{frame}{Ragam 3: Teredam Kurang (\textit{Underdamped}, $\alpha < \omega_0$ atau $\zeta < 1$)}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Karakteristik Fisik:}
    \begin{itemize}
      \item Terjadi jika $R < 2\sqrt{L/C}$ (resistansi redaman kecil).
      \item Diskriminan negatif: $D < 0 \implies$ akar kompleks konjugat:
      \begin{equation*}
        s_{1,2} = -\alpha \pm j \omega_d
      \end{equation*}
      dengan \textbf{Frekuensi Sudut Teredam}:
      \begin{equation*}
        \mathbf{\omega_d = \sqrt{\omega_0^2 - \alpha^2} = \omega_0 \sqrt{1 - \zeta^2}}
      \end{equation*}
      \item \textbf{Bentuk Solusi Umum:}
      \begin{equation*}
        \mathbf{i(t) = e^{-\alpha t} (C_1 \cos\omega_d t + C_2 \sin\omega_d t)}
      \end{equation*}
      \item Menghasilkan \textbf{osilasi sinusoidal teredam}.
    \end{itemize}

    \column{0.44\textwidth}
    \begin{center}
      \resizebox{\linewidth}{!}{%
      \begin{tikzpicture}[scale=0.85]
        \draw[->] (0,-1.8) -- (0,2.2) node[above] {$i(t)$};
        \draw[->] (0,0) -- (4.5,0) node[right] {$t$};
        % Selubung
        \draw[dashed, gray] plot[domain=0:4.2] ({\x}, {2*exp(-0.7*\x)});
        \draw[dashed, gray] plot[domain=0:4.2] ({\x}, {-2*exp(-0.7*\x)});
        % Kurva Underdamped
        \draw[domain=0:4.2, samples=120, smooth, red!85!black, very thick] plot ({\x}, {2*exp(-0.7*\x)*cos(4.5*\x r)});
        \node[red!85!black, font=\scriptsize\bfseries] at (2.8, 1.2) {Underdamped};
      \end{tikzpicture}%
      }
    \end{center}
  \end{columns}
\end{frame}

% FRAME 11: Tangki Osilator LC & Analisis Ruang Fasa
\begin{frame}{Kasus Khusus Tangki LC Murni \& Analisis Ruang Fasa}
  \begin{columns}[c]
    \column{0.52\textwidth}
    Jika $R = 0 \implies \alpha = 0$, PDB tereduksi:
    \begin{align*}
      v_C(t) &= V_m \cos(\omega_0 t + \phi) \\
      i(t) &= -\omega_0 C V_m \sin(\omega_0 t + \phi)
    \end{align*}
    \begin{alertblock}{Kekekalan Energi Abadi}
      Energi total bernilai konstan sepanjang waktu:
      \begin{equation*}
        E_{\text{total}} = \frac{1}{2} C v_C^2 + \frac{1}{2} L i^2 = \mathbf{\frac{1}{2} C V_m^2 = \text{Konstan}}
      \end{equation*}
    \end{alertblock}
    \begin{exampleblock}{Tipologi Ruang Fasa $[v_C, v_C']^T$}
      \tiny
      $\alpha = 0$: Pusat eliptik tertutup. $\zeta < 1$: Fokus spiral memusar masuk. $\zeta \ge 1$: Simpul stabil meluncur langsung ke titik asal $(0,0)$.
    \end{exampleblock}

    \column{0.46\textwidth}
    \begin{center}
      \resizebox{0.88\linewidth}{!}{%
      \begin{tikzpicture}[scale=0.75]
        \draw[->] (-2.2,0) -- (2.2,0) node[right] {$v_C$};
        \draw[->] (0,-2.2) -- (0,2.2) node[above] {$i$};
        \draw[blue!90!black, very thick] (0,0) ellipse (1.8 and 1.2);
        \node[blue!90!black, font=\scriptsize] at (1.1,1.3) {Orbit Tertutup};
        \node[font=\tiny] at (0,-1.6) {Pusat Eliptik (\textit{Center Point})};
      \end{tikzpicture}%
      }
    \end{center}
  \end{columns}
\end{frame}

% FRAME 12: Worked Example RLC
\begin{frame}{Contoh Soal Terhitung (Worked Example): RLC Seri Industri}
  \textbf{Kasus Filter Industri:} $L = 0{,}5\text{ H}$, $C = 20\,\mu\text{F}$. Kapasitor mula-mula bermuatan $v_C(0) = 100\text{ V}$, $i(0) = 0$.
  \vspace{0.1cm}

  \begin{columns}[t]
    \column{0.5\textwidth}
    \textbf{1. Frekuensi Alami \& Resistansi Kritis:}
    \begin{align*}
      \omega_0 &= \frac{1}{\sqrt{LC}} = \frac{1}{\sqrt{0{,}5 \times 20\mu\text{F}}} = \mathbf{316{,}2\text{ rad/s}} \\
      R_{\text{kritis}} &= 2\sqrt{\frac{L}{C}} = 2\sqrt{\frac{0{,}5}{20\mu\text{F}}} = \mathbf{316{,}2\ \Omega}
    \end{align*}
    \textbf{2. Skenario Underdamped ($R = 60\ \Omega$):}
    \begin{align*}
      \alpha &= \frac{R}{2L} = \frac{60}{2(0{,}5)} = \mathbf{60\text{ Np/s}} \\
      \omega_d &= \sqrt{\omega_0^2 - \alpha^2} = \sqrt{316{,}2^2 - 60^2} = \mathbf{310{,}5\text{ rad/s}}
    \end{align*}

    \column{0.5\textwidth}
    \textbf{3. Solusi Khusus Tegangan Kapasitor:}
    Dengan $v_C(0) = 100\text{ V}$ dan $i(0) = 0$:
    \begin{align*}
      C_1 &= 100\text{ V} \\
      C_2 &= \frac{\alpha}{\omega_d} C_1 = \frac{60}{310{,}5}(100) = 19{,}32\text{ V}
    \end{align*}
    \begin{equation*}
      \mathbf{v_C(t) = e^{-60t} (100 \cos 310{,}5t + 19{,}3 \sin 310{,}5t)}
    \end{equation*}
    \textbf{4. Periode Osilasi Teredam:}
    \begin{equation*}
      T_d = \frac{2\pi}{\omega_d} = \frac{2\pi}{310{,}5} \approx \mathbf{20{,}2\text{ ms}}
    \end{equation*}
  \end{columns}
\end{frame}

function rlc!(du, u, p, t)
  zeta, w0 = p
  du[1] = u[2]

end

u0 = [1.0, 0.0]; tspan = (0.0, 0.05)

  \begin{exampleblock}{Intisari Perkuliahan Minggu 4}
    \begin{itemize}
      \item Sistem orde dua dicirikan oleh interaksi dua elemen reaktif penyimpan energi ($L$ dan $C$) dengan persamaan karakteristik kuadrat $s^2 + 2\alpha s + \omega_0^2 = 0$.
      \item Nilai rasio redaman $\zeta = \alpha/\omega_0$ membagi respon menjadi tiga ragam: \textit{Overdamped} ($\zeta > 1$), \textit{Critically Damped} ($\zeta = 1$), dan \textit{Underdamped} ($\zeta < 1$).
      \item Pada tangki LC tanpa rugi-rugi ($R = 0$), energi berosilasi bolak-balik secara kekal abadi, membentuk orbit elips tertutup pada bidang ruang fasa.
    \end{itemize}
