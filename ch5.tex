\section{Struktur Solusi PDB Non-Homogen}
\begin{frame}{Prinsip Superposisi \& Solusi Total}
  \begin{block}{Bentuk Umum PDB Orde 2 Non-Homogen}
    $$ a y'' + b y' + c y = g(x) $$
  \end{block}
  \pause
  \begin{alertblock}{Teorema Solusi Total}
    Solusi lengkap $y(x)$ selalu merupakan penjumlahan dari dua bagian:
    $$ {\color{unibblue} y(x) = y_h(x) + y_p(x)} $$
    \begin{itemize}
      \item $y_h(x)$ (\textit{Homogeneous Solution}): Solusi dari $a y'' + b y' + c y = 0$ (respons alami/transien).
      \item $y_p(x)$ (\textit{Particular Solution}): Solusi khusus yang memenuhi ruas kanan $g(x)$ (respons paksa/keadaan mantap).
    \end{itemize}
  \end{alertblock}
\end{frame}

\section{Metode Koefisien Tak Tentu}
\begin{frame}{Tabel Pemilihan Tebakan Solusi Partikular $y_p(x)$}
  \begin{table}[]
    \centering
    \small
    \begin{tabular}{|l|l|}
      \hline
      \textbf{Bentuk Ruas Kanan $g(x)$} & \textbf{Tebakan Solusi Partikular $y_p(x)$} \\ \hline
      Polinomial $P_n(x) = a_n x^n + \dots + a_0$ & $A_n x^n + A_{n-1} x^{n-1} + \dots + A_0$ \\ \hline
      Eksponensial $e^{kx}$ & $A e^{kx}$ \\ \hline
      Sinusoidal $\cos(\omega x)$ atau $\sin(\omega x)$ & $A \cos(\omega x) + B \sin(\omega x)$ \\ \hline
      Kombinasi $e^{kx}\sin(\omega x)$ & $e^{kx}(A \cos(\omega x) + B \sin(\omega x))$ \\ \hline
    \end{tabular}
  \end{table}
  \pause
  \begin{alertblock}{Aturan Modifikasi (Resonansi)}
    Jika tebakan $y_p(x)$ mengandung suku yang sudah muncul pada $y_h(x)$, kalikan tebakan dengan $x$ (atau $x^2$) hingga tidak ada suku yang berulang!
  \end{alertblock}
\end{frame}

\begin{frame}{Contoh Soal: Koefisien Tak Tentu}
  \textbf{Soal:} Tentukan solusi umum dari $y'' + 3y' + 2y = 4e^{3x}$.
  \pause
  \begin{block}{Langkah 1: Cari Solusi Homogen $y_h(x)$}
    Karakteristik: $r^2 + 3r + 2 = 0 \implies (r+1)(r+2) = 0 \implies r_1 = -1, r_2 = -2$.
    $$ y_h(x) = C_1 e^{-x} + C_2 e^{-2x} $$
  \end{block}
  \pause
  \begin{block}{Langkah 2: Cari Solusi Partikular $y_p(x)$}
    Tebak $y_p = A e^{3x} \implies y_p' = 3A e^{3x}, \quad y_p'' = 9A e^{3x}$.
    \pause
    Substitusikan:
    $9A e^{3x} + 3(3A e^{3x}) + 2(A e^{3x}) = 4e^{3x} \implies 20A e^{3x} = 4e^{3x} \implies A = \frac{4}{20} = \frac{1}{5}$.
  \end{block}
  \pause
  \begin{alertblock}{Solusi Lengkap}
    $$ {\color{unibblue} y(x) = C_1 e^{-x} + C_2 e^{-2x} + \frac{1}{5}e^{3x}} $$
  \end{alertblock}
\end{frame}

\section{Pemodelan Lengkap RLC Seri}
\begin{frame}{Rangkaian RLC Seri dengan Sumber Sinusoidal}
  \begin{block}{Persamaan KVL Rangkaian}
    $$ L \frac{d^2 i}{dt^2} + R \frac{di}{dt} + \frac{1}{C} i = \frac{dV(t)}{dt} = V_m \omega \cos(\omega t) $$
  \end{block}
  \pause
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Parameter Rangkaian:}
      \begin{itemize}
        \item Koefisien Redaman: $\alpha = \frac{R}{2L}$
        \item Frekuensi Resonansi: $\omega_0 = \frac{1}{\sqrt{LC}}$
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \pause
      \textbf{Tiga Rezim Redaman:}
      \begin{itemize}
        \item $\alpha < \omega_0$: \textit{Underdamped}
        \item $\alpha = \omega_0$: \textit{Critically Damped}
        \item $\alpha > \omega_0$: \textit{Overdamped}
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Verifikasi dengan Julia: RLC Seri Interaktif}
  \begin{block}{Simulasi Interaktif}
    Pada praktikum Julia Minggu 5:
    \begin{itemize}
      \item Menyelesaikan sistem keadaan (*state-space*) menggunakan algoritma Runge-Kutta 4 (RK4).
      \item Membandingkan ketiga kurva redaman secara visual.
      \item Mengamati fenomena resonansi listrik saat $\omega = \omega_0$.
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}{Rangkuman}
  \begin{enumerate}
    \item Solusi PDB non-homogen adalah kombinasi linear dari solusi komplementer/homogen dan solusi partikular.
    \item Resonansi terjadi saat frekuensi pemaksa sama dengan frekuensi natural sistem.
    \item Perilaku transien dikendalikan oleh akar-akar homogen, sementara perilaku tunak jangka panjang dikendalikan oleh solusi partikular.
  \end{enumerate}
\end{frame}
