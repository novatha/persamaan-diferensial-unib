\section{PDB Orde 2 Non-Homogen}
\begin{frame}{PDB Orde 2 Non-Homogen}
  \begin{block}{Definisi}
    PDB orde 2 non-homogen memiliki bentuk:
    \begin{equation}
        a y'' + b y' + c y = f(x)
    \end{equation}
    di mana $f(x) \neq 0$.
  \end{block}

  \begin{exampleblock}{Struktur Solusi Umum}
    Solusi umum $y(x)$ terdiri dari penjumlahan dua bagian:
    \begin{equation}
        y(x) = y_h(x) + y_p(x)
    \end{equation}
    \begin{itemize}
        \item $y_h(x)$: Solusi homogen dari persamaan $a y'' + b y' + c y = 0$.
        \item $y_p(x)$: Solusi partikular yang memenuhi persamaan non-homogen.
    \end{itemize}
  \end{exampleblock}
\end{frame}

\section{Metode Koefisien Tak Tentu}
\begin{frame}{Metode Koefisien Tak Tentu}
    Metode koefisien tak tentu digunakan untuk mencari solusi partikular $y_p(x)$ dengan menebak bentuk solusi berdasarkan fungsi $f(x)$.
    
    \begin{alertblock}{Bentuk Tebakan $y_p(x)$}
    \begin{itemize}
        \item Jika $f(x) = k$ (konstan), maka $y_p(x) = A$
        \item Jika $f(x)$ polinomial derajat $n$, maka $y_p(x) = A_n x^n + \dots + A_1 x + A_0$
        \item Jika $f(x) = k e^{\alpha x}$, maka $y_p(x) = A e^{\alpha x}$
        \item Jika $f(x) = k \sin(\beta x)$ atau $k \cos(\beta x)$, maka $y_p(x) = A \cos(\beta x) + B \sin(\beta x)$
    \end{itemize}
    \end{alertblock}
    \textit{Catatan:} Jika tebakan $y_p(x)$ mengandung suku yang sama dengan $y_h(x)$, maka tebakan harus dikalikan dengan $x$ atau $x^2$.
\end{frame}

\section{Aplikasi: Rangkaian RLC Seri}
\begin{frame}{Aplikasi Rangkaian RLC Seri}
    \begin{columns}
        \column{0.4\textwidth}
        \begin{center}
        \begin{tikzpicture}[scale=0.7, transform shape, thick]
            \draw (0,0) -- (0,2) -- (1,2);
            % Resistor
            \draw (1,2) node[above]{R} -- (1.1,2.2) -- (1.3,1.8) -- (1.5,2.2) -- (1.7,1.8) -- (1.9,2.2) -- (2,2);
            \draw (2,2) -- (2.5,2);
            % Inductor
            \draw (2.5,2) node[above]{L} arc (180:0:0.25) arc (180:0:0.25) arc (180:0:0.25);
            \draw (4,2) -- (4.5,2);
            % Capacitor
            \draw (4.5,2) node[above]{C} -- (4.5,2.3);
            \draw (4.7,2) -- (4.7,2.3);
            \draw (4.7,2) -- (5.5,2) -- (5.5,0) -- (0,0);
            % Source
            \draw (0,1) circle (0.3) node {$E$};
            \draw (0,0) -- (0,0.7);
            \draw (0,1.3) -- (0,2);
        \end{tikzpicture}
        \end{center}
        \column{0.6\textwidth}
        Dengan KVL, persamaan rangkaian RLC seri adalah:
        \begin{equation*}
            L \frac{di}{dt} + R i + \frac{1}{C} \int i \, dt = E(t)
        \end{equation*}
        Diferensiasikan terhadap waktu $t$:
        \begin{equation}
            L \frac{d^2i}{dt^2} + R \frac{di}{dt} + \frac{1}{C} i = E'(t)
        \end{equation}
        Ini adalah PDB Orde 2 Non-Homogen!
    \end{columns}
\end{frame}
