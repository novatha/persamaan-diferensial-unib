\section{Definisi Transformasi Laplace}
\begin{frame}{Mengapa Menggunakan Domain Laplace ($s$)?}
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \textbf{Kelemahan Metode Klasik:}
      \begin{itemize}
        \item Menyelesaikan PDB butuh mencari $y_h$, lalu $y_p$, lalu memasukkan IVP.
        \item Rumit untuk sumber tegangan impulsif atau gelombang tangga (*step function*).
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \pause
      \textbf{Keunggulan Laplace:}
      \begin{itemize}
        \item Mengubah operasi kalkulus (turunan/integral) menjadi \textbf{aljabar biasa}.
        \item Syarat awal (IVP) langsung masuk di awal penurunan aljabar!
      \end{itemize}
    \end{column}
  \end{columns}
  \pause
  \begin{block}{Definisi Integral Laplace}
    Untuk fungsi $f(t)$ yang terdefinisi pada $t \ge 0$:
    $$ {\color{unibblue} F(s) = \mathcal{L}\{f(t)\} = \int_{0}^{\infty} e^{-st} f(t) \, dt} $$
    dengan $s = \sigma + j\omega$ adalah variabel frekuensi kompleks.
  \end{block}
\end{frame}

\section{Transformasi Fungsi-Fungsi Dasar}
\begin{frame}{Penurunan Fungsi Elementer}
  \begin{block}{1. Fungsi Konstan: $f(t) = 1$}
    $$ \mathcal{L}\{1\} = \int_{0}^{\infty} e^{-st} dt = \left[ -\frac{1}{s} e^{-st} \right]_{0}^{\infty} = 0 - \left(-\frac{1}{s}\right) = {\color{unibblue} \frac{1}{s}} \quad (s > 0) $$
  \end{block}
  \pause
  \begin{block}{2. Fungsi Eksponensial: $f(t) = e^{at}$}
    $$ \mathcal{L}\{e^{at}\} = \int_{0}^{\infty} e^{-st} e^{at} dt = \int_{0}^{\infty} e^{-(s-a)t} dt = {\color{unibblue} \frac{1}{s - a}} \quad (s > a) $$
  \end{block}
  \pause
  \begin{block}{3. Fungsi Pangkat: $f(t) = t^n$}
    $$ \mathcal{L}\{t^n\} = {\color{unibblue} \frac{n!}{s^{n+1}}} \quad (n \in \mathbb{N}, s > 0) $$
  \end{block}
\end{frame}

\begin{frame}{Tabel Standar Transformasi Laplace}
  \begin{table}[]
    \centering
    \begin{tabular}{|c|c||c|c|}
      \hline
      $f(t)$ & $F(s) = \mathcal{L}\{f(t)\}$ & $f(t)$ & $F(s) = \mathcal{L}\{f(t)\}$ \\ \hline
      $1$ & $\frac{1}{s}$ & $t^n$ & $\frac{n!}{s^{n+1}}$ \\ \hline
      $e^{at}$ & $\frac{1}{s-a}$ & $\cos(\omega t)$ & $\frac{s}{s^2 + \omega^2}$ \\ \hline
      $\sin(\omega t)$ & $\frac{\omega}{s^2 + \omega^2}$ & $\cosh(at)$ & $\frac{s}{s^2 - a^2}$ \\ \hline
      $\sinh(at)$ & $\frac{a}{s^2 - a^2}$ & $e^{at}\sin(\omega t)$ & $\frac{\omega}{(s-a)^2 + \omega^2}$ \\ \hline
    \end{tabular}
  \end{table}
\end{frame}

\section{Sifat-Sifat Penting Transformasi Laplace}
\begin{frame}{Teorema Pergeseran Pertama (s-Shifting)}
  \begin{block}{Teorema Pergeseran Sumbu $s$}
    Jika $\mathcal{L}\{f(t)\} = F(s)$, maka:
    $$ {\color{unibblue} \mathcal{L}\{e^{at} f(t)\} = F(s - a)} $$
  \end{block}
  \pause
  \textbf{Contoh Soal:} Tentukan $\mathcal{L}\{e^{-2t} \cos(3t)\}$.
  \begin{itemize}
    \item Fungsi dasar: $f(t) = \cos(3t) \implies F(s) = \frac{s}{s^2 + 3^2} = \frac{s}{s^2 + 9}$. \pause
    \item Dengan $a = -2$, ganti setiap $s$ dengan $(s - (-2)) = (s + 2)$:
    $$ \mathcal{L}\{e^{-2t} \cos(3t)\} = {\color{unibblue} \frac{s + 2}{(s + 2)^2 + 9}} = \frac{s + 2}{s^2 + 4s + 13} $$
  \end{itemize}
\end{frame}

\begin{frame}{Transformasi Turunan Fungsi (Kunci Penyelesaian PDB)}
  \begin{alertblock}{Teorema Transformasi Turunan}
    \begin{itemize}
      \item Turunan Pertama:
      $$ {\color{unibblue} \mathcal{L}\{f'(t)\} = s F(s) - f(0)} $$
      \item Turunan Kedua:
      $$ {\color{unibblue} \mathcal{L}\{f''(t)\} = s^2 F(s) - s f(0) - f'(0)} $$
    \end{itemize}
  \end{alertblock}
  \pause
  \textit{Catatan: Kondisi awal $f(0)$ dan $f'(0)$ langsung terintegrasi secara otomatis saat menerapkan transformasi!}
\end{frame}

\section{Kuis \& Rangkuman}
\begin{frame}{Kuis Cepat Minggu 6}
  \textbf{Soal:} Tentukan $\mathcal{L}\{t^2 e^{5t}\}$.
  \pause
  \begin{block}{Solusi:}
    \begin{enumerate}
      \item $\mathcal{L}\{t^2\} = \frac{2!}{s^3} = \frac{2}{s^3}$. \pause
      \item Geser $s \to s - 5$:
      $$ \mathcal{L}\{t^2 e^{5t}\} = {\color{unibblue} \frac{2}{(s-5)^3}} $$
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Rangkuman Materi Minggu 6}
  \begin{enumerate}
    \item Transformasi Laplace memetakan fungsi domain waktu $t$ ke aljabar kompleks domain $s$.
    \item Sifat pergeseran pertama mengalikan fungsi dengan eksponensial $e^{at} \iff$ menggeser frekuensi $s-a$.
    \item Transformasi turunan merupakan jembatan utama untuk mengubah PDB menjadi persamaan aljabar.
  \end{enumerate}
\end{frame}
