% FRAME 4: Metode Pecahan Parsial Real Sederhana
\begin{frame}{Metode 1: Kutub Real Sederhana (Metode Penutupan Heaviside)}
  \begin{block}{Dekomposisi Kutub Sederhana}
    Jika penyebut $D(s) = (s - p_1)(s - p_2)\dots(s - p_n)$ memiliki akar-akar real berbeda:
    \begin{equation}
      F(s) = \frac{A_1}{s - p_1} + \frac{A_2}{s - p_2} + \dots + \frac{A_n}{s - p_n}
    \end{equation}
  \end{block}
  \vspace{0.15cm}
  \textbf{Rumus Penutupan Heaviside (\textit{Cover-Up Method}):}
  Koefisien residu $A_k$ dihitung dengan mengalikan $F(s)$ dengan faktor $(s - p_k)$ lalu mengevaluasinya pada $s = p_k$:
  \begin{equation}
    \mathbf{A_k = \lim_{s \to p_k} (s - p_k) F(s)}
  \end{equation}
  Invers domain waktu suku demi suku:
  \begin{equation*}
    \mathbf{f(t) = \left( A_1 e^{p_1 t} + A_2 e^{p_2 t} + \dots + A_n e^{p_n t} \right) u(t)}
  \end{equation*}
\end{frame}

% FRAME 5: Metode Pecahan Parsial Kutub Berulang
\begin{frame}{Metode 2: Kutub Real Berulang (Metode Derivasi Bertingkat)}
  Jika terdapat faktor berulang berorde $m$, yaitu $(s - p)^m$:
  \begin{equation}
    F(s) = \frac{B_m}{(s - p)^m} + \frac{B_{m-1}}{(s - p)^{m-1}} + \dots + \frac{B_1}{s - p} + \dots
  \end{equation}
  \begin{columns}[t]
    \column{0.5\textwidth}
    \begin{block}{Koefisien Orde Tertinggi ($B_m$)}
      Dihitung langsung via cover-up:
      \begin{equation*}
        \mathbf{B_m = \left. (s - p)^m F(s) \right|_{s = p}}
      \end{equation*}
    \end{block}

    \column{0.5\textwidth}
    \begin{alertblock}{Koefisien Orde Lebih Rendah}
      Dihitung melalui turunan bertingkat:
      \begin{equation*}
        \mathbf{B_k = \frac{1}{(m - k)!} \left. \frac{d^{m-k}}{ds^{m-k}} [(s - p)^m F(s)] \right|_{s = p}}
      \end{equation*}
    \end{alertblock}
  \end{columns}
  \vspace{0.2cm}
  Invers Laplace menghasilkan suku polinomial waktu: $\mathcal{L}^{-1}\left\{\frac{1}{(s - p)^k}\right\} = \frac{t^{k-1}}{(k-1)!} e^{pt}$.
\end{frame}

% FRAME 6: Metode Kutub Kompleks Konjugat
\begin{frame}{Metode 3: Kutub Kompleks Konjugat (Melengkapkan Kuadrat)}
  Jika penyebut memuat suku kuadrat definit positif: $s^2 + 2\alpha s + \omega_0^2 = (s + \alpha)^2 + \omega_d^2$:
  \begin{equation}
    F(s) = \frac{A s + B}{(s + \alpha)^2 + \omega_d^2}
  \end{equation}
  Manipulasi pembilang agar sesuai dengan bentuk baku cosinus dan sinus teredam:
  \begin{align*}
    F(s) &= \frac{A(s + \alpha) + (B - A\alpha)}{(s + \alpha)^2 + \omega_d^2} \\
         &= A \frac{s + \alpha}{(s + \alpha)^2 + \omega_d^2} + \frac{B - A\alpha}{\omega_d} \frac{\omega_d}{(s + \alpha)^2 + \omega_d^2}
  \end{align*}
  \begin{exampleblock}{Invers Domain Waktu}
    \begin{equation*}
      \mathbf{f(t) = e^{-\alpha t} \left[ A \cos(\omega_d t) + \frac{B - A\alpha}{\omega_d} \sin(\omega_d t) \right] u(t)}
    \end{equation*}
  \end{exampleblock}
\end{frame}

% FRAME 7: Teorema Konvolusi
\begin{frame}{Teorema Konvolusi Domain Waktu}
  \begin{block}{Teorema Konvolusi}
    Perkalian dua fungsi pada domain frekuensi kompleks $s$ ekuivalen dengan operasi \textbf{integral konvolusi} pada domain waktu:
    \begin{equation}
      \mathbf{\mathcal{L}^{-1}\{F(s) \cdot G(s)\} = (f * g)(t) = \int_0^t f(\tau) g(t - \tau) d\tau}
    \end{equation}
  \end{block}
  \vspace{0.15cm}
  \begin{columns}[c]
    \column{0.52\textwidth}
    \textbf{Makna Fisik dalam Rekayasa Listrik:}
    \begin{itemize}
      \item $H(s)$: Fungsi transfer sistem (\textit{transfer function}).
      \item $X(s)$: Sinyal masukan luar (\textit{input excitation}).
      \item Keluaran sistem $Y(s) = H(s) X(s)$ diperoleh di domain waktu sebagai konvolusi antara respon impuls $h(t)$ dengan sinyal masukan $x(t)$.
    \end{itemize}

    \column{0.44\textwidth}
    \begin{alertblock}{Sifat Komutatif Konvolusi}
      \begin{equation*}
        f(t) * g(t) = g(t) * f(t)
      \end{equation*}
      Sangat ampuh jika dekomposisi pecahan parsial fungsi pendorong sulit difaktorkan!
    \end{alertblock}
  \end{columns}
\end{frame}

% FRAME 8: Teorema Nilai Awal & Akhir (IVT & FVT)
\begin{frame}{Teorema Nilai Awal (IVT) \& Teorema Nilai Akhir (FVT)}
  Mengevaluasi nilai sesaat $t \to 0^+$ dan nilai tunak $t \to \infty$ langsung dari $F(s)$ tanpa perlu melakukan inversi:
  \vspace{0.15cm}
  \begin{columns}[t]
    \column{0.5\textwidth}
    \begin{block}{Teorema Nilai Awal (IVT)}
      \begin{equation}
        \mathbf{f(0^+) = \lim_{s \to \infty} s F(s)}
      \end{equation}
      Berguna untuk memverifikasi nilai arus/tegangan tepat setelah pensaklaran.
    \end{block}

    \column{0.5\textwidth}
    \begin{alertblock}{Teorema Nilai Akhir (FVT)}
      \begin{equation}
        \mathbf{f(\infty) = \lim_{s \to 0} s F(s)}
      \end{equation}
      \textbf{Syarat Mutlak:} Seluruh kutub dari $sF(s)$ harus berada di sebelah kiri bidang kompleks ($\text{Re}(s) < 0$).
    \end{alertblock}
  \end{columns}
  \vspace{0.2cm}
  \footnotesize Peringatan: FVT tidak berlaku jika sistem berosilasi tanpa redaman (kutub di sumbu imajiner) atau tidak stabil!
\end{frame}

% FRAME 10: ZIR vs ZSR
\begin{frame}{Pemisahan Solusi: Respon Masukan-Nol (ZIR) \& Keadaan-Nol (ZSR)}
  Solusi total rangkaian di domain-$s$ selalu terurai secara alami menjadi dua komponen independen:
  \begin{equation}
    \mathbf{Y(s) = Y_{\text{ZIR}}(s) + Y_{\text{ZSR}}(s)}
  \end{equation}
  \begin{columns}[t]
    \column{0.5\textwidth}
    \begin{block}{1. Respon Masukan-Nol (ZIR)}
      \begin{itemize}
        \item Dihasilkan murni oleh \textbf{energi awal} yang tersimpan pada induktor $i(0^-)$ dan kapasitor $v_C(0^-)$.
        \item Seluruh sumber eksitasi luar dimatikan ($V_{\text{sumber}} = 0$).
        \item Menggambarkan peluruhan alami energi internal sistem.
      \end{itemize}
    \end{block}

    \column{0.5\textwidth}
    \begin{exampleblock}{2. Respon Keadaan-Nol (ZSR)}
      \begin{itemize}
        \item Dihasilkan murni oleh \textbf{sumber eksitasi luar}.
        \item Seluruh energi awal diasumsikan nol ($i(0^-)=0, v_C(0^-)=0$).
        \item Terdiri dari kombinasi transien penyalaan dan respon keadaan tunak.
      \end{itemize}
    \end{exampleblock}
  \end{columns}
\end{frame}

% FRAME 11: Studi Kasus TRV Pemutus PLN
\begin{frame}{Studi Kasus Industri: Transient Recovery Voltage (IEC 62271-100)}
  \begin{columns}[c]
    \column{0.55\textwidth}
    \textbf{Fenomena TRV pada Gardu Induk 150 kV PLN:}
    \begin{itemize}
      \item Saat pemutus tenaga (CB) memutus arus hubung singkat pada $t=0$ (arus nol), tegangan di celah kontak melonjak dengan frekuensi tinggi:
      \begin{align*}
        V_{\text{TRV}}(s) &= \frac{V_m \omega_n^2}{s (s^2 + \omega_n^2)} \\
        \mathbf{v_{\text{TRV}}(t)} &= \mathbf{V_m [1 - \cos(\omega_n t)]}
      \end{align*}
      \item \textbf{Amplitudo Puncak Ganda}: Pada $\omega_n t = \pi$:
      \begin{equation*}
        \mathbf{V_{\text{TRV, maks}} = 2 V_m}
      \end{equation*}
      \item Laju Kenaikan Tegangan: $\text{RRRV} = V_m \omega_n = \frac{V_m}{\sqrt{L_s C_s}}$.
    \end{itemize}

    \column{0.45\textwidth}
    \begin{alertblock}{Bahaya Re-ignition}
      Jika RRRV melampaui laju pemulihan dielektrik gas $\text{SF}_6$, busur api menyala kembali (\textit{re-strike}), memicu ledakan katastrofik pada PMT!
    \end{alertblock}
  \end{columns}
\end{frame}

% FRAME 12: Worked Example Invers Laplace
\begin{frame}{Contoh Soal Terhitung (Worked Example): Inversi Kutub Ganda}
  \textbf{Soal:} Tentukan $y(t) = \mathcal{L}^{-1}\{Y(s)\}$ untuk $Y(s) = \frac{2s^2 + 10s + 11}{(s + 1)(s + 2)^2}$.
  \vspace{0.15cm}

  \begin{columns}[t]
    \column{0.48\textwidth}
    \textbf{1. Bentuk Pecahan Parsial:}
    \begin{equation*}
      Y(s) = \frac{A}{s + 1} + \frac{B_2}{(s + 2)^2} + \frac{B_1}{s + 2}
    \end{equation*}
    \textbf{2. Hitung Residu $A$ dan $B_2$:}
    \begin{align*}
      A &= \left. \frac{2s^2 + 10s + 11}{(s + 2)^2} \right|_{s = -1} = \frac{3}{1^2} = \mathbf{3} \\
      B_2 &= \left. \frac{2s^2 + 10s + 11}{s + 1} \right|_{s = -2} = \frac{-1}{-1} = \mathbf{1}
    \end{align*}

    \column{0.48\textwidth}
    \textbf{3. Hitung Residu $B_1$ via Derivasi:}
    \begin{align*}
      B_1 &= \left. \frac{d}{ds} \left[ \frac{2s^2 + 10s + 11}{s + 1} \right] \right|_{s = -2} \\
          &= \left. \frac{(4s+10)(s+1) - (2s^2+10s+11)}{(s+1)^2} \right|_{s=-2} \\
          &= \frac{(2)(-1) - (-1)}{(-1)^2} = \mathbf{-1}
    \end{align*}
    \textbf{4. Solusi Domain Waktu:}
    \begin{equation*}
      \mathbf{y(t) = \left( 3 e^{-t} + t e^{-2t} - e^{-2t} \right) u(t)}
    \end{equation*}
  \end{columns}
\end{frame}
