% FRAME 4: Taksonomi & Klasifikasi PDP Orde Dua
% -------------------------------------------------------------
\begin{frame}{Taksonomi \& Klasifikasi PDP Linier Orde Dua}
  Bentuk umum PDP linier orde dua dengan dua variabel bebas $(x,y)$ atau $(x,t)$:
  \begin{equation*}
    A \frac{\partial^2 u}{\partial x^2} + B \frac{\partial^2 u}{\partial x \partial y} + C \frac{\partial^2 u}{\partial y^2} + D \frac{\partial u}{\partial x} + E \frac{\partial u}{\partial y} + F u = G
  \end{equation*}
  Sifat matematis dan fenomena fisik ditentukan oleh \textbf{Diskriminan}: $\Delta = B^2 - 4AC$.
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \begin{block}{Hiperbolik: $\Delta > 0$}
        \small
        \textbf{Fisika:} Perambatan gelombang \& getaran akustik/EM.\\
        \textbf{Memori:} Kecepatan hingga $v$, syarat batas Dirichlet/Neumann.\\
        \textit{Contoh:} Pers. Gelombang Telegrafer $\frac{\partial^2 v}{\partial x^2} = LC \frac{\partial^2 v}{\partial t^2}$.
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Parabolik: $\Delta = 0$}
        \small
        \textbf{Fisika:} Proses difusi \& transien disipatif tak bolak-balik.\\
        \textbf{Memori:} Relaksasi menuju kesetimbangan termal.\\
        \textit{Contoh:} Difusi Panas Kabel $k \frac{\partial^2 T}{\partial x^2} = \rho c \frac{\partial T}{\partial t}$.
      \end{alertblock}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Elliptik: $\Delta < 0$}
        \small
        \textbf{Fisika:} Distribusi medan tunak/kesetimbangan statik.\\
        \textbf{Memori:} Bergantung penuh pada seluruh batas tertutup.\\
        \textit{Contoh:} Pers. Laplace Medan $\frac{\partial^2 V}{\partial x^2} + \frac{\partial^2 V}{\partial y^2} = 0$.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 5: Teorema Ortogonalitas Fungsi Trigonometri
% -------------------------------------------------------------
\begin{frame}{Fondasi Matematika: Teorema Ortogonalitas}
  \begin{block}{Konsep Ruang Vektor Fungsi}
    Sama seperti vektor basis saling tegak lurus ($\mathbf{i} \cdot \mathbf{j} = 0$), himpunan fungsi $\{1, \cos\frac{n\pi t}{L}, \sin\frac{n\pi t}{L}\}$ saling \textbf{ortogonal} pada selang periodik $[-L, L]$ terhadap perkalian titik berupa integral:
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Relasi Ortogonalitas Sinus \& Kosinus}
        Untuk bilangan bulat positif $m, n \ge 1$:
        \begin{align*}
          \int_{-L}^{L} \cos\left(\frac{m\pi t}{L}\right) \sin\left(\frac{n\pi t}{L}\right) dt &= 0 \\
          \int_{-L}^{L} \cos\left(\frac{m\pi t}{L}\right) \cos\left(\frac{n\pi t}{L}\right) dt &= \begin{cases} 0, & m \ne n \\ L, & m = n \end{cases} \\
          \int_{-L}^{L} \sin\left(\frac{m\pi t}{L}\right) \sin\left(\frac{n\pi t}{L}\right) dt &= \begin{cases} 0, & m \ne n \\ L, & m = n \end{cases}
        \end{align*}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Makna Rekayasa \& Ekstraksi Sinyal}
        \begin{itemize}
          \item Sifat ortogonalitas memungkinkan kita mengekstraksi \textbf{hanya satu komponen frekuensi} tertentu tanpa tercampur harmonisa lainnya.
          \item Merupakan prinsip dasar kerja \textit{Discrete Fourier Transform} (DFT), filter bank harmonisa, dan relai diferensial proteksi modern.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 6: Deret Fourier Trigonometrik & Rumus Euler-Fourier
% -------------------------------------------------------------
\begin{frame}{Formulasi Deret Fourier Trigonometrik}
  \begin{block}{Teorema Representasi Fourier ($T = 2L$)}
    \small
    Sinyal periodik $f(t)$ memenuhi syarat Dirichlet diuraikan menjadi gelombang harmonik:
    \begin{equation*}
      f(t) = a_0 + \sum_{n=1}^{\infty} \left[ a_n \cos\left(\frac{n\pi t}{L}\right) + b_n \sin\left(\frac{n\pi t}{L}\right) \right]
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Koefisien Euler-Fourier}
        \small
        $a_0 = \frac{1}{2L} \int_{-L}^{L} f(t) \, dt \quad \text{(Rata-rata / DC)}$\\[2pt]
        $a_n = \frac{1}{L} \int_{-L}^{L} f(t) \cos\left(\frac{n\pi t}{L}\right) dt$\\[2pt]
        $b_n = \frac{1}{L} \int_{-L}^{L} f(t) \sin\left(\frac{n\pi t}{L}\right) dt$
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Bentuk Polar Magnitudo-Fasa}
        \small
        Sering digunakan dalam sistem tenaga listrik:
        \begin{equation*}
          f(t) = C_0 + \sum_{n=1}^{\infty} C_n \cos(n\omega_0 t - \theta_n)
        \end{equation*}
        $C_n = \sqrt{a_n^2 + b_n^2}$, $\quad \theta_n = \arctan(b_n / a_n)$.
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% -------------------------------------------------------------
% FRAME 7: Pemanfaatan Simetri Gelombang
% -------------------------------------------------------------
\begin{frame}{Pemanfaatan Sifat Simetri Bentuk Gelombang}
  Memeriksa sifat simetri gelombang sebelum mengintegralkan menghemat $\ge 50\%$ waktu komputasi analitis:
  \vspace{0.1cm}
  \begin{columns}[t]
    \begin{column}{0.31\textwidth}
      \begin{block}{Simetri Genap: $f(-t) = f(t)$}
        \small
        \begin{itemize}
          \item Simetris terhadap sumbu-$y$.
          \item Suku sinus hilang: $b_n = 0$.
          \item $a_0 = \frac{1}{L} \int_{0}^{L} f(t) \, dt$
          \item $a_n = \frac{2}{L} \int_{0}^{L} f(t) \cos\left(\frac{n\pi t}{L}\right) dt$
        \end{itemize}
        \textit{Contoh: Penyearah gelombang penuh, kosinus murni.}
      \end{block}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{alertblock}{Simetri Ganjil: $f(-t) = -f(t)$}
        \small
        \begin{itemize}
          \item Simetris terhadap titik $(0,0)$.
          \item Rata-rata nol: $a_0 = 0$.
          \item Suku kosinus hilang: $a_n = 0$.
          \item $b_n = \frac{2}{L} \int_{0}^{L} f(t) \sin\left(\frac{n\pi t}{L}\right) dt$
        \end{itemize}
        \textit{Contoh: Gelombang kotak inverter, gigi gergaji.}
      \end{alertblock}
    \end{column}
    \begin{column}{0.31\textwidth}
      \begin{exampleblock}{Half-Wave: $f(t - L) = -f(t)$}
        \small
        \begin{itemize}
          \item Bentuk paruh kedua pembalikan paruh pertama.
          \item Seluruh harmonisa genap lenyap: $a_{2k} = b_{2k} = 0$.
          \item Hanya memuat harmonisa ganjil ($n = 1, 3, 5, \dots$).
        \end{itemize}
        \textit{Sangat krusial dalam sistem tenaga 3-fasa simetris!}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 9: Contoh Soal Terhitung Langkah demi Langkah
% -------------------------------------------------------------
\begin{frame}{Contoh Terhitung: Deret Fourier Gelombang Persegi $V_0 = 100\text{ V}$}
  \begin{exampleblock}{Masalah: Sinyal Inverter $v(t) = -100\text{ V}$ ($-\pi < t < 0$) \& $+100\text{ V}$ ($0 < t < \pi$)}
    Tentukan koefisien Fourier, ekspansi deret, dan estimasi $\text{THD}_V$ hingga harmonisa ke-7!
  \end{exampleblock}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Penurunan Analitis Koefisien}
        \small
        \textbf{1. Analisis Simetri:}
        $v(-t) = -v(t) \implies \text{Ganjil} \implies a_0 = 0, \, a_n = 0$.\\
        \textbf{2. Integrasi Koefisien Sinus ($L = \pi$):}
        \begin{align*}
          b_n &= \frac{2}{\pi} \int_{0}^{\pi} 100 \sin(nt) \, dt = \frac{200}{\pi} \left[ -\frac{\cos(nt)}{n} \right]_{0}^{\pi} \\
              &= \frac{200}{n\pi} [1 - (-1)^n] = \begin{cases} \mathbf{\dfrac{400}{n\pi}}, & n \text{ ganjil} \\ 0, & n \text{ genap} \end{cases}
        \end{align*}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Sintesis \& Estimasi Distorsi THD}
        \small
        \textbf{3. Ekspansi Deret Tegangan:}
        \begin{equation*}
          v(t) = \frac{400}{\pi} \left[ \sin(t) + \frac{1}{3}\sin(3t) + \frac{1}{5}\sin(5t) + \dots \right]
        \end{equation*}
        \textbf{4. Amplitudo 4 Harmonisa:}
        $V_1 = 127{,}32\text{ V}, \; V_3 = 42{,}44\text{ V}, \; V_5 = 25{,}46\text{ V}, \; V_7 = 18{,}19\text{ V}$.\\[2pt]
        \textbf{5. Distorsi Tegangan Total:}
        \begin{equation*}
          \text{THD}_{V,\le 7} = \frac{\sqrt{V_3^2 + V_5^2 + V_7^2}}{V_1} \times 100\% = \mathbf{41{,}41\%}
        \end{equation*}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 10: Fenomena Gibbs & Konvergensi Titik Diskontinuitas
% -------------------------------------------------------------
\begin{frame}{Fenomena Gibbs (\textit{Gibbs Phenomenon}) pada Tepi Sinyal}
  \begin{columns}[t]
    \begin{column}{0.52\textwidth}
      \begin{alertblock}{Teorema \& Batas Overshoot Asimtotik}
        \small
        Pada setiap diskontinuitas loncat $\Delta f$, jumlahan deret Fourier $S_N(t)$ selalu memiliki overshoot:
        \begin{align*}
          \lim_{N \to \infty} \text{Overshoot} &= \frac{\Delta f}{\pi} \int_{0}^{\pi} \frac{\sin x}{x} \, dx - \frac{\Delta f}{2} \\
          &\approx \mathbf{8{,}95\%} \times \Delta f
        \end{align*}
      \end{alertblock}
      \begin{block}{Konvergensi Nilai Rata-rata Dirichlet}
        \small
        Tepat pada titik patahan loncat $t_0$:
        \begin{equation*}
          f_{\text{Fourier}}(t_0) = \frac{f(t_0^+) + f(t_0^-)}{2}
        \end{equation*}
      \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{exampleblock}{Dampak Rekayasa Elektro}
        \small
        \begin{itemize}
          \item \textbf{Switching IGBT/MOSFET:} Fenomena Gibbs memicu tegangan \textit{ringing} parasitik melampaui rating $V_{\text{DS,max}}$.
          \item \textbf{Bahaya Breakdown:} Tegangan lonjak ini dapat memicu tembus dielektrik pada lilitan transformator.
          \item \textbf{Mitigasi:} Pemasangan snubber $RC$ dan filter perata (\textit{windowing} Lanczos/Hann).
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

% FRAME 11: Analisis Kualitas Daya & Distorsi Harmonisa Total (THD)
% -------------------------------------------------------------
\begin{frame}{Analisis Polusi Harmonisa Sistem Tenaga Listrik}
  \begin{block}{Dekomposisi Arus Beban Nonlinier}
    \small
    Beban industri (VSD, UPS, charger EV) menarik arus tak sinusoidal:
    \begin{equation*}
      i(t) = I_0 + \sqrt{2} I_{1,\text{rms}} \sin(\omega t - \phi_1) + \sum_{n=2}^{\infty} \sqrt{2} I_{n,\text{rms}} \sin(n\omega t - \phi_n)
    \end{equation*}
  \end{block}
  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Definisi Matematis THD}
        \small
        Distorsi arus total:
        \begin{equation*}
          \text{THD}_I = \frac{\sqrt{\sum_{n=2}^{\infty} I_{n,\text{rms}}^2}}{I_{1,\text{rms}}} \times 100\%
        \end{equation*}
        Rugi tembaga: $R_{\text{ac}} \propto \sqrt{n}$, rugi arus pusar: $P_{\text{eddy}} \propto n^2$.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Dampak Kerusakan Jaringan PLN}
        \small
        \begin{itemize}
          \item \textbf{Overheating Trafo:} Rugi panas melonjak.
          \item \textbf{Ledakan Kapasitor Bank:} Resonansi harmonika paralel $n \omega_0 L_{\text{t}} = \frac{1}{n \omega_0 C_{\text{b}}}$.
          \item \textbf{Trip Palsu:} Harmonisa ke-3, 9 menumpuk di kawat netral sistem 3-fasa.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Standar Industri: Regulasi Harmonisa IEEE Std 519-2022}
  \begin{block}{Tanggung Jawab Bersama pada Point of Common Coupling (PCC)}
    \textbf{IEEE Std 519-2022} menetapkan bahwa konsumen bertanggung jawab membatasi distorsi arus ($\text{THD}_I$), sedangkan penyedia utilitas (PLN) membatasi distorsi tegangan ($\text{THD}_V$).
  \end{block}
  \vspace{0.1cm}
  \begin{table}[h]
    \centering
    \scriptsize
    \begin{tabular}{lccc}
      \toprule

x = range(-pi, pi, length=400)
fourier_sq(x, N) = sum((4.0 / (pi * n)) .* sin.(n .* x) for n in 1:2:N)

  \begin{columns}[t]
    \begin{column}{0.48\textwidth}
      \begin{block}{Rangkuman Inti Minggu 9}
        \begin{itemize}
          \item \textbf{PDP vs PDB:} PDP mengatur fenomena terdistribusi ruang-waktu ($u_{tt}, u_t, \nabla^2 u$).
          \item \textbf{Ortogonalitas:} Landasan matematis ekstraksi koefisien Fourier Euler ($a_0, a_n, b_n$).
          \item \textbf{Simetri:} Memanfaatkan simetri genap/ganjil/half-wave mempercepat analisis harmonika.
          \item \textbf{Aplikasi Nyata:} Fenomena Gibbs \& mitigasi distorsi THD sesuai \textbf{IEEE Std 519-2022}.
        \end{itemize}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Pratinjau Minggu 10: Pemisahan Variabel}
        \begin{itemize}
          \item Bagaimana memecahkan PDP secara analitis?
          \item \textbf{Metode Pemisahan Variabel (\textit{Separation of Variables}):} Mengasumsikan solusi $u(x,t) = X(x) \cdot T(t)$.
          \item Mengubah 1 PDP menjadi 2 PDB berkopel melalui konstanta separasi ($-\lambda$).
          \item Deret Fourier menjadi kunci akhir pemenuhan Syarat Awal \& Batas (BVP)!
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}
\end{frame}
