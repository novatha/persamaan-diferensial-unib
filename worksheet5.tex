\section*{Soal-Soal Latihan}

\begin{enumerate}[label=\textbf{\arabic*.}]
    \item \textbf{(C1 - Mengingat)} Tuliskan persamaan diferensial orde 2 non-homogen yang mendeskripsikan arus $i(t)$ pada rangkaian RLC seri!
    \vspace{2.5cm}

    \item \textbf{(C2 - Memahami)} Dalam konteks rangkaian listrik, jelaskan apa yang dimaksud dengan respons transien dan respons tunak (steady-state), dan hubungkan konsep tersebut dengan $y_h(t)$ dan $y_p(t)$.
    \vspace{3.5cm}

    \item \textbf{(C3 - Mengaplikasikan)} Tentukan solusi umum dari persamaan diferensial berikut: 
    $$ y'' - 4y = 8x^2 $$
    \vspace{3.5cm}

    \item \textbf{(C4 - Menganalisis)} Analisislah persamaan differensial $y'' + 9y = 5\sin(3x)$. Temukan solusi partikular $y_p(x)$ menggunakan aturan modifikasi metode koefisien tak tentu!
    \vspace{4.5cm}

    \item \textbf{(C5 - Mengevaluasi)} Diberikan persamaan tegangan kapasitor pada rangkaian seri RC dengan sumber AC: $R \frac{dq}{dt} + \frac{1}{C}q = E_m \cos(\omega t)$. Evaluasi dan bandingkan pendekatan penyelesaian persamaan diferensial orde 1 non-homogen dengan metode fasor yang Anda pelajari di Rangkaian Listrik!
    \vspace{5.5cm}

    \item \textbf{(C6 - Mencipta)} Rancanglah sebuah sistem peredam kejut kendaraan yang memiliki gaya luar sinusoidal dari jalan bergelombang. Formulasikan persamaannya sehingga terjadi fenomena "resonansi". Tentukan parameter yang tepat dan buktikan dengan solusi matematis!
    \vspace{6cm}
\end{enumerate}
